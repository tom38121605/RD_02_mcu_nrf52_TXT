    CC2（6 点闹钟响） → group（总闸） →  set_reset_start_ppi（这条独立线路） → GPIO SET（开灯）

    ppi_channel_include_in_group( ppi, group)

   必须先把每一条 PPI 通道的【EEP 事件、TEP 任务】全部配置完成，再把通道加入 group。

//通过触发task拉高引脚电平，通过普通gpio拉高引脚电平(both)
adc_sensors_intermittent_gpiote_pin_set(  pin)
 
    uint64_t pin_mask = (1<<pin);

    //通过触发task拉高引脚电平
    if(pin_mask & adc_sensors_intermittent_pin_gpiote) 
        nrfx_gpiote_set_task_trigger(pin);
    
    //通过普通gpio拉高引脚电平
    nrf_gpio_pin_set(pin);

//通过触发task拉低引脚电平，通过普通gpio拉低引脚电平(both)
 adc_sensors_intermittent_gpiote_pin_clear(  pin)
 
    uint64_t pin_mask = (1<<pin);

    //通过触发task拉低引脚电平
    if(pin_mask & adc_sensors_intermittent_pin_gpiote) 
        nrfx_gpiote_clr_task_trigger(pin);
    
    //通过普通gpio拉低引脚电平
    nrf_gpio_pin_clear(pin);

//把task设置的读缓冲里面高电平/低电平，备份到普通gpio的写缓冲里面去；从gpiote里收回到普通gpio 
adc_sensors_intermittent_gpiote_pin_disable_gpiote(  pin)

    uint64_t pin_mask = (1ul<<pin);


    //把task设置的读缓冲里面高电平/低电平，备份到普通gpio的写缓冲里面去
    nrf_gpio_pin_write(pin, nrf_gpio_pin_read(pin));

    //从gpiote里收回到普通gpio （先忽略跳过）
    if(pin_mask & adc_sensors_intermittent_pin_gpiote) 
       nrfx_gpiote_out_uninit(pin);

    //清除gpiote里该引脚的标志
    adc_sensors_intermittent_pin_gpiote &= ~(1<<pin);

// 把task设置的读缓冲里面高电平/低电平，备份到普通gpio的写缓冲里面去；从普通 GPIO 交接给 GPIOTE 硬件控制

adc_sensors_intermittent_gpiote_pin_enable_gpiote( pin )
 

    //读出刚才的电平
    bool pin_value = nrf_gpio_pin_read(pin);

    // 设置mux0,mux1 为 toggle，初始为刚才读出的电平
    adc_sensors_intermittent_pin_gpiote_config.init_state = (pin_value ? NRF_GPIOTE_INITIAL_VALUE_HIGH : NRF_GPIOTE_INITIAL_VALUE_LOW);
    nrfx_gpiote_out_init(pin, &adc_sensors_intermittent_pin_gpiote_config);

    //把读缓冲里面的高/低电平，备份到普通gpio的写缓冲里面
    nrf_gpio_pin_write(pin, pin_value);

    //设置引脚为task模式
    nrfx_gpiote_out_task_enable(pin);

    //置位gpiote里该引脚的标志
    adc_sensors_intermittent_pin_gpiote |= 1<<pin;

            channel = channel_port_alloc_low_power(pin, evt_handler, context);

                //pin_configured_set(pin);
                  nrf_bitmask_bit_set(pin, m_cb.configured_pins);   //m_cb.configured_pins | 1<<pin   
            

            if (p_config->hi_accuracy)    //yes 
       
                //高精度时，直接用引脚触发falling中断
                nrf_gpiote_event_configure( channel, pin, p_config->sense );     
                  NRF_GPIOTE->CONFIG[idx] |= pin << GPIOTE_CONFIG_PSEL_Pos | p_config->sense << GPIOTE_CONFIG_POLARITY_Pos;   //  15<<8 | 2<<16 ,  pin, falling
       
            else      
                //低精度时，应是端口触发中断 （先忽略跳过）      
                m_cb.port_handlers_pins[channel - GPIOTE_CH_NUM] |= (p_config->sense) <<  POLARITY_FIELD_POS;
            
                    
            //增加电流解决52840 Errata 155的问题
            //https://docs.nordicsemi.com/bundle/errata_nRF52840_EngA/page/ERR/nRF52840/EngineeringA/latest/anomaly_840_155.html
            if (apply_fix_for_fast_signal_transitions)            
                *(volatile uint32_t *)(NRF_GPIOTE_BASE + 0x600 + (4 * channel)) = 1;            
        
        else        
            err_code = NRFX_ERROR_NO_MEM;

参数：P1.15， evt_handler(null)，  null
channel_port_alloc_event(  pin,  handler,   context)
 
    int8_t   channel_id = NO_CHANNELS;   //-1  //ff

    for (i = 0; i < GPIOTE_CH_NUM; i++)
     
        if (m_cb.handlers[i] == FORBIDDEN_HANDLER_ADDRESS)   // FFFF FFFF  //假如i=6
         
            //pin_in_use_by_te_set(pin, i, handler, context, true);
                m_cb.pin_assignments[15] = channel_id;
                m_cb.handlers[6] = handler;
                m_cb.handlers_context[channel_id] = context;

            channel_id = i;   //6


            //SET_BIT(m_cb.event_channels, i);
              m_cb.event_channels |= i<<1;

            break;


        else if (!IS_SET(m_cb.event_channels, i))  //如果最小的可用通道关联的是task（不是event），则退出  （因为task是从上往下遍历的，说明上面的座位都被task占满了）
         
            break;

    return channel_id;  //6

//使能 GPIOTE 外设的中断 （房间总开关），使能 GPIOTE 的 PORT 事件中断 （小灯子插排开关）

nrfx_gpiote_init( )

    if (m_cb.state != NRFX_DRV_STATE_UNINITIALIZED)  //m_cb是GPIOE的结构体
        return err_code;


    //把所有存在的 GPIO 引脚都标记为‘空闲可用  （只是防止gpio的外部中断自己重复，固即使p0.7已配置了为RXD引脚也没关系）  //相当于班主任的献血登记表里的空闲，跟学生短跑跳高无关
    for (i = 0; i < MAX_PIN_NUMBER; i++)   //42    

        if (nrf_gpio_pin_present_check(i))  // 检查引脚是否存在      
  
           //pin_in_use_clear(i);
             m_cb.pin_assignments[pin] = PIN_NOT_USED;


    // 清空回调函数，清空上下文
    for (i = 0; i < (GPIOTE_CH_NUM + NRFX_GPIOTE_CONFIG_NUM_OF_LOW_POWER_EVENTS); i++)   // 8 + 1    

        //channel_free(i);
            m_cb.handlers[i]                =  FORBIDDEN_HANDLER_ADDRESS;   // 清空回调函数
            m_cb.handlers_context[i]        =  NULL;                        // 清空上下文

            if (8)
               m_cb.port_handlers_pins[8 - 8]  =  PIN_NOT_USED;     // 如果是低功耗通道，清空引脚号


    //把 GPIOTE 驱动中所有已配置引脚的记录全部清零
    memset(m_cb.configured_pins, 0, sizeof(m_cb.configured_pins));
    m_cb.event_channels = 0;

    //设置 GPIOTE 外设的中断优先级为6
    NRFX_IRQ_PRIORITY_SET(nrfx_get_irq_number(NRF_GPIOTE), NRFX_GPIOTE_CONFIG_IRQ_PRIORITY);

    //使能 GPIOTE 外设的中断 （房间总开关）
    NRFX_IRQ_ENABLE(nrfx_get_irq_number(NRF_GPIOTE)); 

    //清除 GPIOTE PORT 引脚变化外部中断事件标志位
    nrf_gpiote_event_clear(NRF_GPIOTE_EVENTS_PORT);

    //使能 GPIOTE 的 PORT 事件中断 （小灯插排开关）
    nrf_gpiote_int_enable(GPIOTE_INTENSET_PORT_Msk);


    m_cb.state = NRFX_DRV_STATE_INITIALIZED;
