参数：this, event_queue_start_tx_pending, event_queue_start_tx_process, nullptr, nullptr, true

event_queue_manager::register_task_impl( context,  pending_func,  process_func,  acquire_func,  release_func,  main_loop_only)
 
    i = num_registered_functions;   //1  （前面有一个usb rx的任务）

    function_context[1]     = this;
    process_functions[1]    = event_queue_start_tx_process;
    pending_functions[1]    = event_queue_start_tx_pending;
    acquire_functions[1]    = nullptr;
    release_functions[1]    = nullptr;
    main_loop_only_flags[i] = true;       // 仅在主循环中运行 

    num_registered_functions++;           //2， 注册的总任务数加 1

event_queue_manager::reset( )

    num_registered_functions = 0;
    process_executing_mask = 0;
    for(uint16_t i=0; i<EVENT_QUEUE_MAX_FUNCTIONS; i++)   //32
    
        acquire_functions[i]    = nullptr;
        release_functions[i]    = nullptr;
        main_loop_only_flags[i] = false;
