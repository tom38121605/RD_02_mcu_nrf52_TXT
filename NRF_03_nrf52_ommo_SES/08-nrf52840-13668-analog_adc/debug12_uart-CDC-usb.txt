
======================= 第一阶段：初始化（建厂挂牌） =======================

main() 
  │
  ├─► USBD_Init() / USB_OS_Init()      // 底层硬件与系统任务初始化
  ├─► NRF_CLOCK->TASKS_HFCLKSTART = 1  // 启动高频时钟
  ├─► USBD_AddDriver()                 // 挂载 nRF52 硬件驱动
  ├─► USBD_EnableIAD()                 // 声明为 USB 复合设备
  │
  ├─► USBD_AddEPEx(&EPBulkIn)          // 申请 Bulk 批量端点（大货车，传 ADC 数据）
  ├─► USBD_AddEPEx(&EPIntIn)           // 申请 Int 中断端点（小电驴，传状态）
  │
  ├─► usbd_main_add_cdc()              // 申请缓冲区，注册 CDC 串口功能（发营业执照）
  ├─► usb_serial_init(received, event) // 注册：数据到达回调、插拔状态回调
  │
  └─► USBD_Start()                     // 拉起 D+ 引脚（开门营业）
        │
        ▼ (电脑枚举，识别出 COM 口)


======================= 第二阶段：数据通讯（流水线运转） =======================

【接收流程：电脑 -> 板子 (RX)】

[电脑] 串口助手发送指令
  │
  ▼ (USB 物理线缆)
[USB 硬件 EP OUT 端点] 
  │
  ▼ (触发 USB 中断，数据存入接收缓冲区)
[协议栈底层] 
  │
  ▼ (自动调用注册好的接收回调)
[usb_serial_packet_received()] (main.cpp)
  │
  ├─► 解析指令 (如 OMMO_COMMAND_USB_DATA_ENABLE)
  └─► 启动 ADC 采样状态机


【发送流程：板子 -> 电脑 (TX)】

[底层采样完成] (触发 sample_set_ready)
  │
  ▼
[usb_serial_write_data_packet_vargs()] 
  │ (打包时间戳、ADC数据)
  ▼
[CDC 发送队列 / 缓冲区]
  │
  ▼ (等待电脑主机发来 IN 令牌)
[USB 硬件 EP IN 端点] (通过 EPBulkIn 通道发送)
  │
  ▼
[电脑] 上位机软件接收并解析 ADC 波形


【插拔事件流程】

[物理 USB 拔插] 
  │
  ▼ 
[USB 硬件/底层驱动]
  │ 
  ▼
[usb_serial_event_handler()] (main.cpp)
  │
  ├─► OPENED： 切换状态，停用无线，准备 USB 传数据
  └─► CLOSED： 切换状态，启用 ESB 无线传输

USB_CDC_HANDLE usbd_main_add_cdc( rx_buffer,   rx_buffer_len,   max_tx_buffer_len )
 
    static const USB_ADD_EP_INFO   EPBulkIn =  
        .MaxPacketSize = max_tx_buffer_len,       // Maximum packet size
        .Interval      = 0,                       // Interval not used for Bulk endpoints.
        .Flags         = 0,                       // Flags not used.
        .InDir         = USB_DIR_IN,              // IN direction (Device to Host)
        .TransferType  = USB_TRANSFER_TYPE_BULK,  // Endpoint type - Bulk.
        .ISO_Type      = USB_ISO_SYNC_TYPE_NONE
    
    const USB_ADD_EP_INFO   EPBulkOut =  
        .MaxPacketSize = rx_buffer_len,            // Maximum packet size
        .Interval      = 0,                     // Interval not used for Bulk endpoints.
        .Flags         = 0,                     // Flags not used.
        .InDir         = USB_DIR_OUT,           // OUT direction (Host to Device)
        .TransferType  = USB_TRANSFER_TYPE_BULK,// Endpoint type - Bulk.
        .ISO_Type      = USB_ISO_SYNC_TYPE_NONE
    

    USB_CDC_INIT_DATA InitData = {};

    InitData.EPIn              = USBD_AddEPEx(&EPBulkIn, nullptr, 0);
    InitData.EPOut             = USBD_AddEPEx(&EPBulkOut, rx_buffer, rx_buffer_len);   //输入端口, rx_buffer, 3072
 
    static const USB_ADD_EP_INFO   EPIntIn =  
        .MaxPacketSize = 16, 
        .Interval      = 64,                   // 8 ms 
        .Flags         = 0,
        .InDir         = USB_DIR_IN,
        .TransferType  = USB_TRANSFER_TYPE_INT,
        .ISO_Type      = USB_ISO_SYNC_TYPE_NONE
    
    InitData.EPInt             = USBD_AddEPEx(&EPIntIn, nullptr, 0);
 
    return USBD_CDC_Add(&InitData);

USBD_Init( )  

  const USB_HW_DRIVER    * pDriver;

   
  //USB_Global，USB_aEPStat置初值
  USB_MEMSET(&USB_Global, 0, sizeof(USB_Global));
  USB_Global.sCopyright = "SEGGER emUSBD V" VERSION2STRING(USB_VERSION);
  USB_Global.NumEPs     = 1;
  USB_Global.USBSpecRev = 0x0200;
  USB_Global.MaxPower   = USB_MAX_POWER;
  USB_Global.MaxPowerSS = USB_MAX_POWER / 4u;
  USB_Global.pfOnBCDVersion = _GetBCDVersion;
  USB_MEMSET(USB_aEPStat, 0, sizeof(USB_aEPStat));
   

  // 待办事项流水线 ，设置变量_aEvent[0] 到 _aEvent[8]都为 0xffffffff
  USB_OS_Init();
   

  //设置usb协议栈底层驱动 
  USBD_X_Config();

      //清零事件标记
      NRF_CLOCK->EVENTS_HFCLKSTARTED = 0;

      //启动“高频时钟”（HFCLK），如没有外部晶振，启动芯片内部的 64MHz RC 振荡器（HFINT）
      NRF_CLOCK->TASKS_HFCLKSTART = 1;

      //等待时钟稳定
      while (NRF_CLOCK->EVENTS_HFCLKSTARTED == 0) { }

      //向 USB 协议栈注册底层硬件驱动（把 nRF52 专属的底层代码“挂载”到系统里）
      //USBD_AddDriver(&USB_Driver_Nordic_nRF52xxx);   
        USB_Global.pDriver =USB_Driver_Nordic_nRF52xxx;

      //向 USB 协议栈注册中断开关
      //USBD_SetISRMgmFuncs(_EnableISR, NULL, NULL);
      //  USBD_SetISREnableFunc(_EnableISR)
          USB_Global.pfEnableISR = _EnableISR;


   
// 直接触发中断函数中的事件  (这里是空宏，意味着关闭事件回调)
  USBD_PROFILE_CALL_VOID(USBD_EVTID_INIT);

  pDriver = USB_Global.pDriver;


  if (USB_Global.pfAttach == NULL)  
    USB_Global.pfAttach = _DummyAttach;

   
  if (pDriver->pfInit != NULL)  
    pDriver->pfInit();


 
  USB_aEPStat[0].MaxPacketSize = pDriver->pfGetMaxPacketSize(0);

  USB_Global.IsInited = 1;

  //空宏，忽略跳过
  USBD_PROFILE_END_CALL(USBD_EVTID_INIT);

USBD_RegisterSCHook(  * pHook,   *pfStateCb,  * pContext) 
 
  USB_HOOK *  p;

  pHook->cb            = pfStateCb;   //static_usb_on_state_change
  pHook->pContext      = pContext;    //this
   

  //判断是否已重复注册，如重复注册则报错，如果没有则插入到链表里面 (从第一个位置开始，看里面是否已有hook)  //因确定是第一次注册，忽略下面5句
  p = USB_Global.pFirstSCHook;  
  while (p != NULL)  
    if (p == pHook)  
      return 1;     // 重复注册则报错     
    p = p->pNext;    
  
  
  pHook->pNext = USB_Global.pFirstSCHook;
  USB_Global.pFirstSCHook = pHook;            //把hook插入到首位

usb_serial_packet_received(uint8_t data[], uint16_t length, const void*)
{
uint16_t response_length = 0;
uint8_t response_buffer[1024];

    //Check for enable packet
    switch(data[0])
    {
        case OMMO_COMMAND_USB_DATA_ENABLE:
            if(data_mode != DATA_MODE_DISABLED || !adc_sensors_is_state_machine_configured())
            {
                response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_MODE);
            }
            else if(length != 2)
            {
                response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_DATA);
            }
            else
            {
                ret_code_t error_code = adc_sensors_enable();
                if(error_code == NRF_SUCCESS)
                {
                    packet_id = data[1];
                    data_mode = DATA_MODE_USB;
                }
                response_length = fill_in_ack_packet(response_buffer, convert_nrf_error_code_to_ommo_ack_code(error_code));

                set_leds();
            }
            break;

        case OMMO_COMMAND_USB_SET_SYNCH_CH_ID:
            if(length != 2)
            {
                response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_DATA);
            }
            else
            {
                response_length = fill_in_ack_packet(response_buffer, convert_nrf_error_code_to_ommo_ack_code(comms_esb_set_synch_rx_channel(data[1])));
            }
            break;
    }

    if(response_length == 0) //No response, try protocol agnostic commands
        response_length = process_packet_received(data, length, response_buffer, sizeof(response_buffer));

    if(response_length == 0) //No response, must be an unknown command
        response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_UNKNOWN_COMMAND);

    usb_serial_send_command_packet(response_buffer, response_length);
}
