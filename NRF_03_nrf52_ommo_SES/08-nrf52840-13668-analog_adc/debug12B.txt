  //static inline void usbd_errata_187_211_begin(void)

      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 3. 关上保险箱（重新上锁）
  
  else  
      *((volatile U32 *)(0x4006ED14)) = 0x00000003;  // 如果保险箱已经是解锁状态，直接打补丁
  

    //修复 USB 唤醒恢复时的时序 Bug（防止唤醒瞬间数据丢失）。  
    //static inline void usbd_errata_171_begin(void)
  if (*((volatile U32 *)(0x4006EC00)) == 0x00000000) // 再次检查保险箱
  
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 塞入暗号 0x9375
      *((volatile U32 *)(0x4006EC14)) = 0x000000C0;  // 在另一个地方打补丁
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 重新上锁
  
  else  
      *((volatile U32 *)(0x4006EC14)) = 0x000000C0;
  

  //开启 USB 外设
  pUSBRegs->ENABLE = 1;
  
  ISB();  //把cpu所有缓冲区里面的指令全部清空

  //static inline void usbd_errata_187_211_end(void)
  if (*((volatile U32 *)(0x4006EC00)) == 0x00000000) // 如果后门保险箱当前没开
