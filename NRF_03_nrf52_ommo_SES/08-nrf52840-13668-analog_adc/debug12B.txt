_IRQHandler( )  


   U32            Status;
   unsigned    i;
   unsigned    EPIndex;
   USB_REGS * pUSBRegs = _pUSBRegs;
   USB_SETUP_PACKET SetupPacket;


   //空函数，忽略跳过
   USBD_ENTER_ISR();


   //看USB物理层（PHY）通电了没，如没通电，就把误报中断全清掉，然后退出中断    
   if ((USBREGSTATUS & OUTPUTRDY) == 0 )  
      
      // PHY is powered down. Ignore all interrupts      
      pUSBRegs->EVENTS_USBEVENT = 0;
      pUSBRegs->EVENTS_USBRESET = 0;
      pUSBRegs->EVENTCAUSE = pUSBRegs->EVENTCAUSE;   //写1清零事件标记
      USB__OnStatusChange(USB_STAT_ATTACHED);        //设备回到 ATTACHED（已连接，但没就绪）状态
      USBD_LEAVE_ISR();
      return;

    
   //看到usb复位的中断，如下处理
   if (pUSBRegs->EVENTS_USBRESET != 0 ) 
      pUSBRegs->EVENTS_USBRESET = {0;
      pUSBRegs->EPDATASTATUS      = pUSBRegs->EPDATASTATUS;   //写1清零事件标记
      pUSBRegs->EVENTS_EPDATA    = 0;
      pUSBRegs->EPINEN               = 0;
      pUSBRegs->EPOUTEN             = 0;
      USB__OnBusReset();
      _SetupActive = 0;





USB__OnBusReset( )  


  //如果收到复位信号时，系统记录里竟然显示‘当前根本没插线（未连接）’， 赶紧往日志里记下log信息
  if ((USB_Global.State & USB_STAT_ATTACHED) == 0 )  
    USB_LOG((USB_MTYPE_CORE, "Bus RESET in unattached state!"));
   

  USB_Global.Addr = 0;

  //把第i号端点的‘死锁/挂起（Halted）’状态强制清零，让它重新活过来，准备好接收或发送数据
  for (i = 1; i < USB_Global.NumEPs; i++) 
    USB_aEPStat[i].IsHalted = 0;

  //只保留设备当前配置的第 0 位（自供电标志），把其他所有特殊状态（比如远程唤醒、USB 3.0 低功耗状态）全部清零
  USB_Global.DeviceState &= 1;

  //把设备状态强制更新为：原来的插上线（ATTACHED）状态 + 准备就绪（READY），并广而告之整个系统
  USB__OnStatusChange((USB_Global.State & USB_STAT_ATTACHED) | USB_STAT_READY);
 







    

   if (pUSBRegs->EVENTS_USBEVENT != 0 )  
      pUSBRegs->EVENTS_USBEVENT = 0;
      Status = pUSBRegs->EVENTCAUSE;
      pUSBRegs->EVENTCAUSE = Status;                     // Clear all bits
      if ((Status & EVENTCAUSE_SUSPEND) != 0u)  
         USB__OnSuspend();
       
      if ((Status & EVENTCAUSE_RESUME) != 0u)  
         USB__OnResume();
       
    

   if (pUSBRegs->EVENTS_EP0SETUP != 0u)  
      pUSBRegs->EVENTS_EP0SETUP = 0;
      SetupPacket.bmRequestType = pUSBRegs->BMREQUESTTYPE;
      SetupPacket.bRequest         = pUSBRegs->BMREQUEST;
      SetupPacket.wValueLow       = pUSBRegs->WVALUEL;
      SetupPacket.wValueHigh      = pUSBRegs->WVALUEH;
      SetupPacket.wIndexLow       = pUSBRegs->WINDEXL;
      SetupPacket.wIndexHigh      = pUSBRegs->WINDEXH;
      SetupPacket.wLengthLow      = pUSBRegs->WLENGTHL;
      SetupPacket.wLengthHigh    = pUSBRegs->WLENGTHH;
      pUSBRegs->EVENTS_EP0DATADONE = 0;
      pUSBRegs->INTENSET                = INT_EP0DATADONE;

      if ((SetupPacket.bmRequestType & 0x80u) == 0 )  
         _SetupActive = -1;
      else  
         _SetupActive = 1;
       
      _EPIN_Busy[0] = 0;
      USB__HandleSetup(&SetupPacket);

      if (_SetupActive < 0)  
         //
         // Issue EP0 read
         //
         pUSBRegs->TASKS_EP0RCVOUT = 1;
       
    

 
   // 旧版本遗留的历史无用垃圾，忽略跳过
   EPIndex = _ExtraPacket_EPIndex;  //_ExtraPacket_EPIndex是旧版本遗留的历史无用垃圾，忽略跳过
   if (EPIndex != 0 )   
      _ExtraPacket_EPIndex = 0;     //忽略跳过
      USB__OnRx(EPIndex, _PacketBuff, _ExtraPacket_NumBytes);


    

   pUSBRegs->EVENTS_STARTED = 0;
   pUSBRegs->INTENCLR          = INT_STARTED;
 
   // EP0-related interrupt occurred
 
   if (pUSBRegs->EVENTS_EP0DATADONE != 0u)  
      pUSBRegs->EVENTS_EP0DATADONE = 0;

      if (_SetupActive > 0)  
         _EPIN_Busy[0] = 0;
         USB__OnTx(0);
       
      if (_SetupActive < 0)  
         _ReadPacketEP0();

         if (_SetupActive < 0)  
             
            // Issue next EP0 read
             
            pUSBRegs->TASKS_EP0RCVOUT = 1;
         
       
      // Any other EPx-related interrupt occurred
 
   pUSBRegs->EVENTS_EPDATA = 0;
   Status = pUSBRegs->EPDATASTATUS;
   if (Status != 0u) {
      pUSBRegs->EPDATASTATUS = Status;      // Clear all bits

      //或已完成发送数据到pc
      for (i = 1; i < NUM_PHY_EP; i++)  
         Status >>= 1;

         if ((Status & 1u) != 0u)  
            _EPIN_Busy[i] = 0;

            if (_EPIN_Enabled[i] != 0)  
               EPIndex = USB__EPAddr2Index(i | 0x80u) ;
               USB__OnTx(EPIndex);
             
          
      //或已完成读取pc发来的数据 
      Status >>= (18u - NUM_PHY_EP);                        // Shift to EPOUT bits
      for (i = 1; i < NUM_PHY_EP; i++)  

         if ((Status & 1u) != 0u)  

            if (_EPOUT_Enabled[i] != 0)  
               EPIndex = USB__EPAddr2Index(i) ;
               _ReadPacket(i, EPIndex);
              else  
               _EPOUT_DataPresent[i] = 1;
             
         
         Status >>= 1;
      
   

   //空函数，忽略跳过
   USBD_LEAVE_ISR();
