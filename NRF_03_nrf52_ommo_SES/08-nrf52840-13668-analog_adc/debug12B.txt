_Start() 

  pUSBRegs =  (USB_REGS *)0x40027000u;

  //关全局中断
  //USB_OS_IncDI();
    NRFX_CRITICAL_SECTION_ENTER();


  //来自 Nordic SDK 的更新版 USB 启用代码 (修复 USB 在电脑休眠唤醒时（挂起状态）容易死机卡死的 Bug)
  //static inline void usbd_errata_187_211_begin(void)
  if (*((volatile U32 *)(0x4006EC00)) == 0x00000000) // 如果后门保险箱没开过（状态是0）
  
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 1. 塞入接头暗号 0x9375，强行解锁！
      *((volatile U32 *)(0x4006ED14)) = 0x00000003;  // 2. 在特定寄存器里打上补丁（修硬件Bug）
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 3. 关上保险箱（重新上锁）
  
  else  
      *((volatile U32 *)(0x4006ED14)) = 0x00000003;  // 如果保险箱已经是解锁状态，直接打补丁
  

    //修复 USB 唤醒恢复时的时序 Bug（防止唤醒瞬间数据丢失）。  
    //static inline void usbd_errata_171_begin(void)
  if (*((volatile U32 *)(0x4006EC00)) == 0x00000000) // 再次检查保险箱
  
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 塞入暗号 0x9375
      *((volatile U32 *)(0x4006EC14)) = 0x000000C0;  // 在另一个地方打补丁
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 重新上锁
  
  else  
      *((volatile U32 *)(0x4006EC14)) = 0x000000C0;
  

  开启 USB 外设
  pUSBRegs->ENABLE = 1;
  
  ISB();  //把cpu所有缓冲区里面的指令全部清空  
  DSB();  //等待。等前面所有的内存读写操作（写寄存器、写内存）彻底、真实地在硬件上生效，才准继续往下走。

  // 等待usb ready，应需要几us 
  while ((pUSBRegs->EVENTCAUSE & EVENTCAUSE_READY) == 0) { }

  pUSBRegs->EVENTCAUSE = EVENTCAUSE_READY;   //1<<11, 清零 usb ready 标记

  ISB();  //把cpu所有缓冲区里面的指令全部清空  
  DSB();  //等待。等前面所有的内存读写操作（写寄存器、写内存）彻底、真实地在硬件上生效，才准继续往下走。


   //刚才往 0x4006EC14 写了 0xC0，现在恢复00，防止这个临时补丁影响后续其他外设
  //static inline void usbd_errata_171_end(void)
  if (*((volatile U32 *)(0x4006EC00)) == 0x00000000) // 如果后门保险箱当前没开（状态是0）
  
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 1. 塞入接头暗号 0x9375，强行解锁！
      *((volatile U32 *)(0x4006EC14)) = 0x00000000;  // 2. 把寄存器清零（解除刚才的打补丁状态）
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 3. 重新上锁
  
  else  
      *((volatile U32 *)(0x4006EC14)) = 0x00000000;  // 如果保险箱本来就开着，直接清空寄存器
  

  //刚才往 0x4006ED14 写了 0x03，现在恢复00，防止这个临时补丁影响后续其他外设
  //static inline void usbd_errata_187_211_end(void)
  if (*((volatile U32 *)(0x4006EC00)) == 0x00000000) // 如果后门保险箱当前没开
  
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 1. 塞入接头暗号 0x9375，强行解锁！
      *((volatile U32 *)(0x4006ED14)) = 0x00000000;  // 2. 把寄存器清零（解除刚才的打补丁状态）
      *((volatile U32 *)(0x4006EC00)) = 0x00009375;  // 3. 重新上锁
  
  else  
      *((volatile U32 *)(0x4006ED14)) = 0x00000000;  // 如果保险箱本来就开着，直接清空寄存器
  

  USB_Global.pfEnableISR(_IRQHandler);

  
  //清零所有端口  
  pUSBRegs->EPINEN          = 0;
  pUSBRegs->EPOUTEN         = 0;
  

  // 清零所有事件标记  
  pUSBRegs->EVENTCAUSE   = pUSBRegs->EVENTCAUSE;
  pUSBRegs->EPDATASTATUS = pUSBRegs->EPDATASTATUS;
  pUSBRegs->EVENTS_EPDATA   = 0;
  pUSBRegs->EVENTS_USBEVENT = 0;
  pUSBRegs->EVENTS_USBRESET = 0;
  

  //使能下面的usb中断  
  pUSBRegs->INTEN = INT_USBRESET
                  | INT_USBEVENT
                  | INT_EP0SETUP
                  | INT_EPDATA
                  ;

  //开全局中断
  //USB_OS_DecRI();
    NRFX_CRITICAL_SECTION_EXIT();
