
timer0：   adc_sampling_timer


TIMER3 (自由运行，从不停)
   │
   │ sample event (每 ~1ms)
   │ (PPI)
   ▼
TIMER0 TASKS_START
   │
   │ CC0 → MUX 切换
   │ CC1 → ADC START 脉冲 (中断)
   │ CC2 → Set/Reset 开始
   │ CC3 → Set/Reset 结束
   │        ├── SHORT STOP  ──► TIMER0 停
   │        └── SHORT CLEAR ──► TIMER0 计数器清零
   │
   ▼
（等下一个 TIMER3 sample event 再次启动 TIMER0）


TIMER3 是节拍源，永远在跑。

TIMER0 是执行器，每轮被 TIMER3 打一拍跑一次，跑完自己停。

TIMER0 停的时候，TIMER3 照常计数、照常产生 sample event。

二、 运行与中断链路图（硬件与软件的交汇）
展示 TIMER0 被触发启动后，硬件 PPI 如何自动执行动作（零 CPU 消耗），以及中断触发后软件状态机如何推进。


时间轴 ──────────────────────────────►

[TIMER3 触发] 
    │
    ▼
TIMER0 TASKS_START (计数器从 0 开始跑)
    │
    │
    ├─► [CC0 触发]  (ADC_MUX_SWITCH_TIME_US)
    │     ├─ (PPI 硬件) 切换 MUX 到 X 通道
    │     └─ (PPI 硬件) 开启 X 桥接，关闭上一通道桥接
    │
    │
    ├─► [CC1 触发]  (ADC_START_SAMPLE_TIME_US)
    │     ├─ (PPI 硬件) 翻转 OMMO_ADC_START_PIN，给外部 ADC 发转换脉冲
    │     ├─ (PPI 硬件) 触发内部 SAADC 采样
    │     └─ (中断软件) 进入 sampling_timer_event_handler()
    │           └─ 推进状态机 (PENDING -> COMPLETE)，配置下一次 CC0/CC1 和 PPI
    │
    │
    │    (外部 ADC 吭哧吭哧转换中... 转换完成拉低 DRDY)
    │     ├─ (PPI 硬件) DRDY 触发 SPIM START，读回 24-bit 数据
    │     └─ (中断软件) SPI END 触发 comm_device_event_handler()
    │           └─ 保存 X 通道数据，原子操作推进状态机，预装下一次 SPI
    │
    │    (TIMER0 继续计数，等待 Y 通道和 Z 通道的重复流程...)
    │    (Y 通道: CC0' -> CC1' -> SPI读)
    │    (Z 通道: CC0'' -> CC1'' -> SPI读)
    │
    │
    ├─► [CC2 触发]  (ADC_SET_RESET_START_TIME_US)
    │     └─ (PPI 硬件) 拉高 Set/Reset 引脚
    │
    │
    └─► [CC3 触发]  (ADC_SET_RESET_START_TIME_US + 1us)
          ├─ (PPI 硬件) 拉低 Set/Reset 引脚
          ├─ (SHORT 硬件) COMPARE3_STOP  ──► 停 TIMER0
          └─ (SHORT 硬件) COMPARE3_CLEAR ──► 清 TIMER0 计数器
    │
    ▼
(本轮结束，等待下一次 TIMER3 触发)


------

三、 总结

从上面的图可以清晰看出 TIMER0 的设计哲学：

初始化阶段：不配中断（除了 CC1），拼命配 PPI 和 SHORTS。把所有“准点该干活”的琐事全交给了硬件。

运行阶段：CC0 只管硬件切通道，CC2/CC3 只管硬件发脉冲并自尽（STOP/CLEAR）。

中断阶段：唯一的 CC1 中断只干“预判”和“修改下一次参数”的活，绝不干涉当前的硬件执行。

这样既保证了 24-bit ADC 要求的微秒级严格时序，又没有给 CPU 带来任何中断负担。

nrfx_timer_init( p_instance, p_config, timer_event_handler )

    timer_control_block_t * p_cb = &m_cb[p_instance->instance_id];


    NRFX_ASSERT(p_config);
    NRFX_ASSERT(timer_event_handler);

    nrfx_err_t err_code;

    if (p_cb->state != NRFX_DRV_STATE_UNINITIALIZED)
    {
        err_code = NRFX_ERROR_INVALID_STATE;
        NRFX_LOG_WARNING("Function: %s, error code: %s.",
                         __func__,
                         NRFX_LOG_ERROR_STRING_GET(err_code));
        return err_code;
    }

    /* Warning 685: Relational operator '<=' always evaluates to 'true'"
     * Warning in NRF_TIMER_IS_BIT_WIDTH_VALID macro. Macro validate timers resolution.
     * Not necessary in nRF52 based systems. Obligatory in nRF51 based systems.
     */

    /*lint -save -e685 */

    NRFX_ASSERT(NRF_TIMER_IS_BIT_WIDTH_VALID(p_instance->p_reg, p_config->bit_width));

    //lint -restore

    p_cb->handler = timer_event_handler;      //sampling_timer_event_handler
    p_cb->context = p_config->p_context;

    uint8_t i;
    for (i = 0; i < p_instance->cc_channel_count; ++i)
    {
        nrf_timer_event_clear(p_instance->p_reg,
                              nrf_timer_compare_event_get(i));
    }

    NRFX_IRQ_PRIORITY_SET(nrfx_get_irq_number(p_instance->p_reg),
        p_config->interrupt_priority);
    NRFX_IRQ_ENABLE(nrfx_get_irq_number(p_instance->p_reg));

    nrf_timer_mode_set(p_instance->p_reg, p_config->mode);
    nrf_timer_bit_width_set(p_instance->p_reg, p_config->bit_width);
    nrf_timer_frequency_set(p_instance->p_reg, p_config->frequency);  //0

    p_cb->state = NRFX_DRV_STATE_INITIALIZED;

    err_code = NRFX_SUCCESS;
    NRFX_LOG_INFO("Function: %s, error code: %s.",
                  __func__,
                  NRFX_LOG_ERROR_STRING_GET(err_code));
    return err_code;
}

__STATIC_INLINE void nrf_timer_frequency_set(NRF_TIMER_Type * p_reg,
                                             nrf_timer_frequency_t frequency)
{
    p_reg->PRESCALER = (p_reg->PRESCALER & ~TIMER_PRESCALER_PRESCALER_Msk) |
                         ((frequency << TIMER_PRESCALER_PRESCALER_Pos) &
                              TIMER_PRESCALER_PRESCALER_Msk);
}

sampling_timer_event_handler(nrf_timer_event_t event_type, void* p_context)
{
    UNUSED_PARAMETER(p_context);


    //Called just after ADC start sample happens while ADC is sampling
    if(event_type == ADC_START_SAMPLE_CC_EVENT)
    {
        // Actions in every case is preparations for next state, X is prepared at beginning.
        // For the rbridge selection the task is used, fork is used for cleaning only.
        switch(adc_sampling_state)
        {
            case ADC_SAMPLING_STATE_X_CHANNEL_PENDING:
            {
                // Set new sample time (Y channel)  
                //写入timer0 CC0的阈值 240 * 1us，不使能timer0 CC0中断，开启timer0， 用于切换到y通道
                nrfx_timer_compare(&adc_sampling_timer, ADC_MUX_SWITCH_CC_CHANNEL, US_TO_TICKS_16MHZ(ADC_MUX_SWITCH_TIME_US(1)), false);      // 240 * 1 

                //写入timer0 CC1的阈值 60+240 * 1us，使能timer0 CC1中断，开启timer0， 用于开启y通道的adc转换
                nrfx_timer_compare(&adc_sampling_timer, ADC_START_SAMPLE_CC_CHANNEL, US_TO_TICKS_16MHZ(ADC_START_SAMPLE_TIME_US(1)), true);   // 60 + 240*1

 
                // Configure MUX select 0b00 (0b01 -> 0b00) for Y channel  //把闹铃PPI连上事件切换引脚电平 
                nrf_ppi_task_endpoint_setup(mux_select_ppi, nrfx_gpiote_clr_task_addr_get(OMMO_ADC_MUX_A1_PIN));
                nrf_ppi_fork_endpoint_setup(mux_select_ppi, nrfx_gpiote_clr_task_addr_get(OMMO_ADC_MUX_A0_PIN));
 

                // Configure bridge select on for Y channel and bridge select off for X channel
                adc_sensors_intermittent_gpiote_pin_disable_gpiote(OMMO_IO_Z_EN);
                adc_sensors_intermittent_gpiote_pin_enable_gpiote(OMMO_IO_Y_EN);
                nrf_ppi_task_endpoint_setup(channel_bridge_select_ppi, nrfx_gpiote_set_task_addr_get(OMMO_IO_Y_EN));
                nrf_ppi_fork_endpoint_setup(channel_bridge_select_ppi, nrfx_gpiote_clr_task_addr_get(OMMO_IO_X_EN));
                adc_sampling_state = ADC_SAMPLING_STATE_X_CHANNEL_COMPLETE;
                break;
            }

            case ADC_SAMPLING_STATE_Y_CHANNEL_PENDING:
            {
                // Set new sample time (Z channel)
                //写入timer0 CC0的阈值 240 * 2us，不使能timer0 CC0中断，开启timer0， 用于切换到z通道
                nrfx_timer_compare(&adc_sampling_timer, ADC_MUX_SWITCH_CC_CHANNEL, US_TO_TICKS_16MHZ(ADC_MUX_SWITCH_TIME_US(2)), false);      // 240 * 2 

                //写入timer0 CC1的阈值 60+240 * 2us，不使能timer0 CC1中断，开启timer0， 用于开启z通道的adc转换
                nrfx_timer_compare(&adc_sampling_timer, ADC_START_SAMPLE_CC_CHANNEL, US_TO_TICKS_16MHZ(ADC_START_SAMPLE_TIME_US(2)), false);   // 60 + 240*2

 
                // Configure MUX select 0b10 (0b00 -> 0b10) for Z channel
                nrf_ppi_task_endpoint_setup(mux_select_ppi, nrfx_gpiote_set_task_addr_get(OMMO_ADC_MUX_A1_PIN));
                nrf_ppi_fork_endpoint_setup(mux_select_ppi, nrfx_gpiote_clr_task_addr_get(OMMO_ADC_MUX_A0_PIN));
 

                // Configure bridge select on for Z channel and bridge select off for Y channel
                adc_sensors_intermittent_gpiote_pin_disable_gpiote(OMMO_IO_X_EN);
                adc_sensors_intermittent_gpiote_pin_enable_gpiote(OMMO_IO_Z_EN);
                nrf_ppi_task_endpoint_setup(channel_bridge_select_ppi, nrfx_gpiote_set_task_addr_get(OMMO_IO_Z_EN));
                nrf_ppi_fork_endpoint_setup(channel_bridge_select_ppi, nrfx_gpiote_clr_task_addr_get(OMMO_IO_Y_EN));
 
                adc_sampling_state = ADC_SAMPLING_STATE_Y_CHANNEL_COMPLETE;
                break;
            }

            case ADC_SAMPLING_STATE_X_CHANNEL_COMPLETE:
            case ADC_SAMPLING_STATE_Y_CHANNEL_COMPLETE:
                // SPI RXEND ISR runs at a lower priority and did not complete in time
                adc_sampling_state = ADC_SAMPLING_STATE_ERROR;
                // Prevent any further interrupts until the state machine resets
                nrfx_timer_compare_int_disable(&adc_sampling_timer, ADC_START_SAMPLE_CC_CHANNEL);
                break;

            case ADC_SAMPLING_STATE_IDLE:
                break;

            default:
                OMMO_APP_ERROR_CHECK(NRF_ERROR_INTERNAL, L(adc_sampling_state));
        }
    }

}
