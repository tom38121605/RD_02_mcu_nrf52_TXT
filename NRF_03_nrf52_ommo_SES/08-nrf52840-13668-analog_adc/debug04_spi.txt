nrfx_spim_init(  const p_instance,   p_config,    handler,   *p_context)

    spim_control_block_t * p_cb = &m_cb[0];

    p_spim =  p_instance->p_reg;    //NRF_SPIM0


    //-----------------spi中断函数设置----------------

    irq_handlers[0] = nrfx_spim_0_irq_handler;
   
    //nrfx_prs_acquire(p_instance->p_reg, irq_handlers[0]);     //无操作，忽略跳过
    //  nrfx_prs_acquire( NRF_SPIM0, nrfx_spim_0_irq_handler );
        无操作;


    p_cb->handler = handler;       //spim_handler ，spi中断回调函数
    p_cb->p_context = p_context;   //this


    //-----------------spi中断函数设置----end------------



    //-----------------spi引脚设置----------------


    if (p_config->mode <= NRF_SPIM_MODE_1)      //1， yes    
        nrf_gpio_pin_clear(p_config->sck_pin);  //sck置低电平

    //设置sck引脚：输出，连接输入缓冲，无上下拉，S0S1标准驱动或2ma，无边缘检测
    nrf_gpio_cfg(p_config->sck_pin,
                 NRF_GPIO_PIN_DIR_OUTPUT,
                 NRF_GPIO_PIN_INPUT_CONNECT,
                 NRF_GPIO_PIN_NOPULL,
                 NRF_GPIO_PIN_S0S1,
                 NRF_GPIO_PIN_NOSENSE);

    //设置MO引脚：输出，初始低电平
    mosi_pin = p_config->mosi_pin;
    nrf_gpio_pin_clear(mosi_pin);
    nrf_gpio_cfg_output(mosi_pin);
 
    //设置MO引脚：输入
    miso_pin = p_config->miso_pin;
    nrf_gpio_cfg_input(miso_pin, (nrf_gpio_pin_pull_t)NRFX_SPIM_MISO_PULL_CFG);
 

    p_cb->miso_pin = p_config->miso_pin;
    p_cb->ss_pin = p_config->ss_pin;

 
    //设置SS引脚：输出，初始高电平
    nrf_gpio_pin_set(p_config->ss_pin);       
    nrf_gpio_cfg_output(p_config->ss_pin);

    p_cb->ss_active_high = p_config->ss_active_high;  //false
 

    //设置spi引脚的外设功能
    //nrf_spim_pins_set(p_spim, p_config->sck_pin, mosi_pin, miso_pin);
      p_reg->PSEL.SCK  = sck_pin;
      p_reg->PSEL.MOSI = mosi_pin;
      p_reg->PSEL.MISO = miso_pin;

    //-----------------spi引脚设置----end------------


    //设置FREQUENCY, spi的频率为16M
    //nrf_spim_frequency_set(p_spim, p_config->frequency);
       p_reg->FREQUENCY =  frequency;                           //16M

    //设置CONFIG, 模式1， MSB
    //nrf_spim_configure(p_spim, p_config->mode, p_config->bit_order);
       p_reg->CONFIG =  0<<0 | 0<<2 | 1<< 1
 
    //nrf_spim_orc_set(p_spim, p_config->orc);
      p_reg->ORC = 0xff;

    //nrf_spim_enable(p_spim);
      p_reg->ENABLE =  7<<0;


    //设置spi的中断优先级为1，使能spi的irq总中断
    if (p_cb->handler)  //spim_handler ，spi中断回调函数

        //设置spi的中断优先级为1
        //NRFX_IRQ_PRIORITY_SET(nrfx_get_irq_number(p_instance->p_reg),   p_config->irq_priority);

        //使能spi的irq总中断
        //NRFX_IRQ_ENABLE(nrfx_get_irq_number(p_instance->p_reg));
          NVIC_EnableIRQ(irq_number);


    p_cb->transfer_in_progress = false;
    p_cb->state = NRFX_DRV_STATE_INITIALIZED;

ret_code_t spim_basic::trx(const uint8_t * p_tx_buffer, const size_t tx_length, uint8_t * p_rx_buffer, const size_t rx_length,
                           const comm_device_event_handler_t handler, void * p_context, const comm_device_trx_flags_t flags)
{
    evt_handler = handler;
    evt_context = p_context;
    trx_complete = false;
    is_repeated = !!(flags & COMM_DEVICE_FLAGS_REPEATED);

    switch (acq)
    {
        case SPIMB_ACQ_NORMAL:
        {
            if (rx_length + tx_length > SPIM_MAX_XFER_SIZE)
            {
                return NRF_ERROR_INVALID_LENGTH;
            }

            //Save output buffer to copy response to after SPI TRX
            evt_rx_buffer = p_rx_buffer;

            // Setup transfer
            nrfx_spim_xfer_desc_t xfer_desc = NRFX_SPIM_XFER_TRX(p_tx_buffer, tx_length, io_buffer, tx_length + rx_length);
            nrf_spim_enable(spim_instance->p_reg);
            return nrfx_spim_xfer(spim_instance, &xfer_desc, cast_flags_to_nrf(flags));
        }

        case SPIMB_ACQ_DIRECT:
        {
            const nrfx_spim_xfer_desc_t xfer = NRFX_SPIM_XFER_TRX(p_tx_buffer, tx_length, p_rx_buffer, rx_length);
            nrf_spim_enable(spim_instance->p_reg);
            return nrfx_spim_xfer(spim_instance, &xfer, cast_flags_to_nrf(flags));
        }

        default:
            return NRF_ERROR_INVALID_STATE;
    }
}

uint32_t nrfx_spim_start_task_get(nrfx_spim_t const * p_instance)
{
    NRF_SPIM_Type * p_spim = (NRF_SPIM_Type *)p_instance->p_reg;
    return nrf_spim_task_address_get(p_spim, NRF_SPIM_TASK_START);  //ADS127L11 DRDY 变低（事件）→ PPI 线路 → 硬件自动调用 SPIM TASK_START → SPI 自动发读数据指令，不用 CPU 干预。
}


typedef enum
{
    NRF_SPIM_TASK_START   = offsetof(NRF_SPIM_Type, TASKS_START),   ///< Start SPI transaction.    //开启 spi传输，这里是读3各adc数据
    NRF_SPIM_TASK_STOP    = offsetof(NRF_SPIM_Type, TASKS_STOP),    ///< Stop SPI transaction.
    NRF_SPIM_TASK_SUSPEND = offsetof(NRF_SPIM_Type, TASKS_SUSPEND), ///< Suspend SPI transaction.
    NRF_SPIM_TASK_RESUME  = offsetof(NRF_SPIM_Type, TASKS_RESUME)   ///< Resume SPI transaction.
} nrf_spim_task_t;
