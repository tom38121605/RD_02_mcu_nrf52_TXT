event_queue_manager::execute_once_impl( in_main )


    //------------------------- 检查是否在硬件中断里 --------------------------
 
    AAA = current_int_priority_get();  //取得当前模块的中断编号 ， APP_IRQ_PRIORITY_THREAD表示在main的while循环中

    if (current_int_priority_get() ！= APP_IRQ_PRIORITY_THREAD )  //如果在其它中断里(不在main中)，则马上退出
       STRING("event_queue called from IRQ");
       return;

    //------------------------- 检查是否在硬件中断里 ----end----------------------


    for(  i=0; i<num_registered_functions; i++)
    
        // 对usb tx，这里的AAA和BBB都是true，固会往下运行，不会略过
        AAA= main_loop_only_flags[1];    //true，这个任务是否只能在主循环执行
        BBB = in_main;                   //true, 现在是不是在主循环里  （可不可以运行分配给主循环的任务）
        if( AAA && !BBB) continue;


        // 如果这个任务 usb tx正在执行中（防止它自己套娃自己），直接跳过   （因为中断已被排除，只可能自己套自己）
        bit = (1 << 1);
        if(process_executing_mask & bit) continue;


        //这个任务 usb tx正在执行
        process_executing_mask |= bit;



        // 检查 pending ，不符合条件的pending会直接跳过
        if(pending_functions[1] != nullptr)   //空pending会总是允许运行
         
            //pending = ((event_queue_task_pending_context)pending_functions[i])(function_context[i]);
              pending = event_queue_start_tx_pending(this);

            //看有没有该任务的flg
            if(!pending)            
                process_executing_mask &= ~bit;
                continue;
            
        

        // Try to acquire resources — null acquire means always allowed.
        if(acquire_functions[i] != nullptr)
        
            bool acquired = ((event_queue_task_acquire)acquire_functions[i])();

            if(!acquired)
             
                process_executing_mask &= ~bit; // resource unavailable — skip; release is not called
                continue;
             
         

        TRACE_WITH_ARGS_2(TRACE_TYPE_TIME_FLASHADDR_U16, TRACE_LOCATION_EVENT_QUEUE_PROCESSING,   process_functions[i], i);



        //运行该任务的服务子程序                               
        //process_functions[i](this);
          event_queue_start_tx_process(this);



        //这个任务 usb tx 已执行完毕
        process_executing_mask &= ~bit;


        //下面不会运行，忽略跳过
        if(release_functions[i] != nullptr)  //no，nullptr
            ((event_queue_task_release)release_functions[i])();

################################ 方法1 ###################################



【中断上下文 (ISR)】ADC 采样完成
  └─> sample_set_ready() [main.cpp]
        └─> usb_serial_write_data_packet_vargs() [usb_serial.hpp 内联]
              └─> write_data_packet_vargslist() [comms_packetizer.cpp]
                    ├─> COBS 编码后写入 tx_buffer_active (主/副缓冲区)
                    └─> (标记数据已就绪，拉高 flush 标志)



════════════════ 上下文切换 (退出中断，回到 main_event_queue) ════════════════

【主循环上下文 (Thread)】
main_event_queue.execute_forever() [main.cpp]
  │
  ├─> 查任务：event_queue_start_tx_pending() [comms_packetizer.cpp] (检查 Tx Buffer 是否有货)
  │     │
  │     ▼
  └─> event_queue_start_tx_process() [comms_packetizer.cpp]
        │
        ├─> 切换双缓冲 (tx_buffer_active 换成 tx_buffer_secondary)
        ├─> 记录本次发送的指针区间 (tx_buffer_transport)
        │
        └─> initiate_tx()
              │
              └─> setup_next_tx()
                    │
                    └─> transport_device->comms_transport_tx() [comms_usb_serial.cpp]
                          │
                          ├─> 填充 write_io_ctx (数据指针、长度、完成回调 static_usb_cdc_on_write_cplt)
                          │
                          └─> USBD_CDC_WriteAsync() [USB_CDC.c]
                                │
                                └─> USBD_WriteAsync() [USB_Main.c]
                                      │ (往队列塞入 DataParts)
                                      └─> USBD_WriteOverlapped()
                                            │
                                            └─> pfStartTx() 
                                                  └─> USB__Send()
                                                        └─> _SendEPEx() [USB_HW_Nordic_nRF.c]
                                                              └─> 启动底层 EasyDMA 发送

════════════════ 硬件发送完毕，触发中断 ════════════════

【中断上下文 (ISR)】USB 发送完成 (DMA 搬运完毕)
  └─> _IRQHandler() [USB_HW_Nordic_nRF.c] (检测到 ENDEPIN)
        └─> USB__OnTx() [USB_Main.c]
              └─> (回调) static_usb_cdc_on_write_cplt() [comms_usb_serial.cpp]
                    │
                    └─> tx_complete_callback() (即 static_transport_write_complete)
                          └─> transport_write_complete() [comms_packetizer.cpp]
                                │
                                ├─> 检查是否全部发完 (tx_buffer_transport 是否到达 end)
                                │     ├─> 若没发完，调 setup_next_tx() 继续发
                                │     └─> 若发完了，调 on_tx_complete() 收尾 (释放标志位)
