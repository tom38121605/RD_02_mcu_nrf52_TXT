======================= 【前置准备：成立物流公司】 =======================

1. USBD_Init() / USBD_X_Config()    -> 购买货车、建仓库（配置底层寄存器）
2. USBD_AddDriver() / _Start()      -> 雇佣专职司机、打通高速路（打硅片补丁，拉起时钟）
3. USBD_CDC_Add(&InitData)          -> 成立 CDC 串口专线（分配端点、建缓冲区）
4. BSP_USB_InstallISR / EnableISR   -> 设立前台接待处（注册中断，接通电话线）
5. USBD_SetOnRxEP0 / LineState      -> 培训前台如何应对客户的特殊要求（注册控制回调）

static void _IRQHandler(void) {
  U32        Status;
  unsigned   i;
  unsigned   EPIndex;
  USB_REGS * pUSBRegs = _pUSBRegs;
  USB_SETUP_PACKET SetupPacket;

  USBD_ENTER_ISR();
  if ((USBREGSTATUS & OUTPUTRDY) == 0u) {
    //
    // PHY is powered down. Ignore all interrupts.
    //
    pUSBRegs->EVENTS_USBEVENT = 0;
    pUSBRegs->EVENTS_USBRESET = 0;
    pUSBRegs->EVENTCAUSE = pUSBRegs->EVENTCAUSE;
    USB__OnStatusChange(USB_STAT_ATTACHED);
    USBD_LEAVE_ISR();
    return;
  }
  if (pUSBRegs->EVENTS_USBRESET != 0u) {
    pUSBRegs->EVENTS_USBRESET = 0;
    pUSBRegs->EPDATASTATUS    = pUSBRegs->EPDATASTATUS;
    pUSBRegs->EVENTS_EPDATA   = 0;
    pUSBRegs->EPINEN          = 0;
    pUSBRegs->EPOUTEN         = 0;
    USB__OnBusReset();
    _SetupActive = 0;
  }
  if (pUSBRegs->EVENTS_USBEVENT != 0u) {
    pUSBRegs->EVENTS_USBEVENT = 0;
    Status = pUSBRegs->EVENTCAUSE;
    pUSBRegs->EVENTCAUSE = Status;              // Clear all bits
    if ((Status & EVENTCAUSE_SUSPEND) != 0u) {
      USB__OnSuspend();
    }
    if ((Status & EVENTCAUSE_RESUME) != 0u) {
      USB__OnResume();
    }
  }
  if (pUSBRegs->EVENTS_EP0SETUP != 0u) {
    pUSBRegs->EVENTS_EP0SETUP = 0;
    SetupPacket.bmRequestType = pUSBRegs->BMREQUESTTYPE;
    SetupPacket.bRequest      = pUSBRegs->BMREQUEST;
    SetupPacket.wValueLow     = pUSBRegs->WVALUEL;
    SetupPacket.wValueHigh    = pUSBRegs->WVALUEH;
    SetupPacket.wIndexLow     = pUSBRegs->WINDEXL;
    SetupPacket.wIndexHigh    = pUSBRegs->WINDEXH;
    SetupPacket.wLengthLow    = pUSBRegs->WLENGTHL;
    SetupPacket.wLengthHigh   = pUSBRegs->WLENGTHH;
    pUSBRegs->EVENTS_EP0DATADONE = 0;
    pUSBRegs->INTENSET           = INT_EP0DATADONE;
    if ((SetupPacket.bmRequestType & 0x80u) == 0u) {
      _SetupActive = -1;
    } else {
      _SetupActive = 1;
    }
    _EPIN_Busy[0] = 0;
    USB__HandleSetup(&SetupPacket);
    if (_SetupActive < 0) {
      //
      // Issue EP0 read
      //
      pUSBRegs->TASKS_EP0RCVOUT = 1;
    }
  }
  //
  // Handle extra packet
  //
  EPIndex = _ExtraPacket_EPIndex;
  if (EPIndex != 0u) {
    _ExtraPacket_EPIndex = 0;
    USB__OnRx(EPIndex, (U8 *)_PacketBuff, _ExtraPacket_NumBytes);
  }
  pUSBRegs->EVENTS_STARTED = 0;
  pUSBRegs->INTENCLR       = INT_STARTED;
  //
  // EP0-related interrupt occurred
  //
  if (pUSBRegs->EVENTS_EP0DATADONE != 0u) {
    pUSBRegs->EVENTS_EP0DATADONE = 0;
    if (_SetupActive > 0) {
      _EPIN_Busy[0] = 0;
      USB__OnTx(0);
    }
    if (_SetupActive < 0) {
      _ReadPacketEP0();
      if (_SetupActive < 0) {
        //
        // Issue next EP0 read
        //
        pUSBRegs->TASKS_EP0RCVOUT = 1;
      }
    }
  }
  //
  // Any other EPx-related interrupt occurred
  //
  pUSBRegs->EVENTS_EPDATA = 0;
  Status = pUSBRegs->EPDATASTATUS;
  if (Status != 0u) {
    pUSBRegs->EPDATASTATUS = Status;              // Clear all bits
    for (i = 1; i < NUM_PHY_EP; i++) {
      Status >>= 1;
      if ((Status & 1u) != 0u) {
        _EPIN_Busy[i] = 0;
        if (_EPIN_Enabled[i] != 0) {
          EPIndex = USB__EPAddr2Index(i | 0x80u) ;
          USB__OnTx(EPIndex);
        }
      }
    }
    Status >>= (18u - NUM_PHY_EP);                // Shift to EPOUT bits
    for (i = 1; i < NUM_PHY_EP; i++) {
      if ((Status & 1u) != 0u) {
        if (_EPOUT_Enabled[i] != 0) {
          EPIndex = USB__EPAddr2Index(i) ;
          _ReadPacket(i, EPIndex);
        } else {
          _EPOUT_DataPresent[i] = 1;
        }
      }
      Status >>= 1;
    }
  }
  USBD_LEAVE_ISR();
}

usb_serial_packet_received(uint8_t data[], uint16_t length, const void*)
{
uint16_t response_length = 0;
uint8_t response_buffer[1024];

    //Check for enable packet
    switch(data[0])
    {
        case OMMO_COMMAND_USB_DATA_ENABLE:
            if(data_mode != DATA_MODE_DISABLED || !adc_sensors_is_state_machine_configured())
            {
                response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_MODE);
            }
            else if(length != 2)
            {
                response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_DATA);
            }
            else
            {
                ret_code_t error_code = adc_sensors_enable();
                if(error_code == NRF_SUCCESS)
                {
                    packet_id = data[1];
                    data_mode = DATA_MODE_USB;
                }
                response_length = fill_in_ack_packet(response_buffer, convert_nrf_error_code_to_ommo_ack_code(error_code));

                set_leds();
            }
            break;

        case OMMO_COMMAND_USB_SET_SYNCH_CH_ID:
            if(length != 2)
            {
                response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_DATA);
            }
            else
            {
                response_length = fill_in_ack_packet(response_buffer, convert_nrf_error_code_to_ommo_ack_code(comms_esb_set_synch_rx_channel(data[1])));
            }
            break;
    }

    if(response_length == 0) //No response, try protocol agnostic commands
        response_length = process_packet_received(data, length, response_buffer, sizeof(response_buffer));

    if(response_length == 0) //No response, must be an unknown command
        response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_UNKNOWN_COMMAND);

    usb_serial_send_command_packet(response_buffer, response_length);
}
