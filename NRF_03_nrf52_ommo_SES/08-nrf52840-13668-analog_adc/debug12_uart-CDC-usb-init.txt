
======================= 第一阶段：初始化（建厂挂牌） =======================

main() 
  │
  ├─► USBD_Init() / USB_OS_Init()      // 底层硬件与系统任务初始化
  ├─► NRF_CLOCK->TASKS_HFCLKSTART = 1  // 启动高频时钟
  ├─► USBD_AddDriver()                 // 挂载 nRF52 硬件驱动
  ├─► USBD_EnableIAD()                 // 声明为 USB 复合设备
  │
  ├─► USBD_AddEPEx(&EPBulkIn)          // 申请 Bulk 批量端点（大货车，传 ADC 数据）
  ├─► USBD_AddEPEx(&EPIntIn)           // 申请 Int 中断端点（小电驴，传状态）
  │
  ├─► usbd_main_add_cdc()              // 申请缓冲区，注册 CDC 串口功能（发营业执照）
  ├─► usb_serial_init(received, event) // 注册：数据到达回调、插拔状态回调
  │
  └─► USBD_Start()                     // 拉起 D+ 引脚（开门营业）
        │
        ▼ (电脑枚举，识别出 COM 口)


======================= 第二阶段：数据通讯（流水线运转） =======================

【接收流程：电脑 -> 板子 (RX)】

[电脑] 串口助手发送指令
  │
  ▼ (USB 物理线缆)
[USB 硬件 EP OUT 端点] 
  │
  ▼ (触发 USB 中断，数据存入接收缓冲区)
[协议栈底层] 
  │
  ▼ (自动调用注册好的接收回调)
[usb_serial_packet_received()] (main.cpp)
  │
  ├─► 解析指令 (如 OMMO_COMMAND_USB_DATA_ENABLE)
  └─► 启动 ADC 采样状态机


【发送流程：板子 -> 电脑 (TX)】

[底层采样完成] (触发 sample_set_ready)
  │
  ▼
[usb_serial_write_data_packet_vargs()] 
  │ (打包时间戳、ADC数据)
  ▼
[CDC 发送队列 / 缓冲区]
  │
  ▼ (等待电脑主机发来 IN 令牌)
[USB 硬件 EP IN 端点] (通过 EPBulkIn 通道发送)
  │
  ▼
[电脑] 上位机软件接收并解析 ADC 波形


【插拔事件流程】

[物理 USB 拔插] 
  │
  ▼ 
[USB 硬件/底层驱动]
  │ 
  ▼
[usb_serial_event_handler()] (main.cpp)
  │
  ├─► OPENED： 切换状态，停用无线，准备 USB 传数据
  └─► CLOSED： 切换状态，启用 ESB 无线传输

comms_usb_serial::static_usb_cdc_on_write_cplt( pIOContext )

    class_ptr = (comms_usb_serial*)pIOContext->pContext;

    USB_TRACE(TRACE_USB_SERIAL_WRITE_COMPLETE);   // 记录进入中断时间 

    switch(pIOContext->Status)
    
    case 0:
        err_code = NRF_SUCCESS;  
    
    case USB_STATUS_EP_HALTED:
    case USB_STATUS_EP_BUSY:
        err_code = NRF_ERROR_BUSY;
 
    case USB_STATUS_ERROR:  // typically means suspended
    default:
        err_code = NRF_ERROR_NOT_FOUND; 
    
    
    class_ptr->tx_complete_callback(pIOContext->NumBytesTransferred, err_code, this );

usbd_main_add_cdc( rx_buffer,   rx_buffer_len,   max_tx_buffer_len )
 
    static const USB_ADD_EP_INFO   EPBulkIn =  
        .MaxPacketSize = max_tx_buffer_len,       // 16384
        .Interval      = 0,                       // Interval not used for Bulk endpoints.
        .Flags         = 0,                       // Flags not used.
        .InDir         = USB_DIR_IN,              // 1
        .TransferType  = USB_TRANSFER_TYPE_BULK,  // Endpoint type - Bulk.
        .ISO_Type      = USB_ISO_SYNC_TYPE_NONE
    
    const USB_ADD_EP_INFO   EPBulkOut =  
        .MaxPacketSize = rx_buffer_len,            // 3072
        .Interval      = 0,                     // Interval not used for Bulk endpoints.
        .Flags         = 0,                     // Flags not used.
        .InDir         = USB_DIR_OUT,           // OUT direction (Host to Device)
        .TransferType  = USB_TRANSFER_TYPE_BULK,// Endpoint type - Bulk.
        .ISO_Type      = USB_ISO_SYNC_TYPE_NONE
    

    USB_CDC_INIT_DATA InitData = {};


    //添加输出到pc端口：批发大货车
    InitData.EPIn              = USBD_AddEPEx(&EPBulkIn, nullptr, 0);                  


    //添加从pc输入端口：批发大货车
    //把接收变量和接收长度保存到结构体 USB_aEPStat
    //把端口编号存到端口地址
    //InitData.EPOut             = USBD_AddEPEx(&EPBulkOut, rx_buffer, rx_buffer_len);   
      InitData.EPOut             = _AddEP(&EPBulkOut, usb_cdc_raw_buffer,  3072);   

 
    static const USB_ADD_EP_INFO   EPIntIn =  
        .MaxPacketSize = 16, 
        .Interval      = 64,                   // 8 ms 
        .Flags         = 0,
        .InDir         = USB_DIR_IN,              // 1
        .TransferType  = USB_TRANSFER_TYPE_INT,
        .ISO_Type      = USB_ISO_SYNC_TYPE_NONE
    
    //添加输出到pc端口：零售小电驴
    InitData.EPInt             = USBD_AddEPEx(&EPIntIn, nullptr, 0);    
 

    //向 USB 设备栈注册一路 CDC ACM 虚拟串口，分配接口、绑定端点
    return USBD_CDC_Add(&InitData);

  USB_CDC_HANDLE    hInst;
  USB_CDC_INST    * pInst;


  // 默认串口状态：SERIAL_STATE（DCD/DSR/CTS/RI 等信号）`
  // aDefaultStatus`：**CDC 通知端点默认状态报文**，主机通过中断 IN 端点读取这个串口信号状态（载波检测、CTS 等）
  static const U8   aDefaultStatus[] =  
    0xA1, 0x20, 0x00, 0x00,
    0x00, 0x00, 0x02, 0x00,
    0x00, 0x00
   ;

  // 检查CDC 串口数量：看看还能不能加串口，超过上限直接死机。
  
  _Init();
  if (_NumInstances >= USB_CDC_DEVICES)  
    USB_PANIC("Too many class interfaces");


  // ========== 第一个接口：CDC CONTROL 控制接口 ==========
  // 接口 1（控制接口）：用 EPInt 端点。专门负责接收电脑发来的配置命令

  InterfaceNum     = USB__AllocIF();
  FirstInterfaceNo = InterfaceNum;
  pInterface                = &USB_Global.aIF[InterfaceNum];
  pInterface->pfAddFuncDesc = _AddFuncDesc;   // 回调：生成CDC Functional描述符
  pInterface->IFClass       = 2;             // 0x02 = CDC Control
  pInterface->IFSubClass    = 2;             // 0x02 = ACM
  pInterface->IFProtocol    = 1;             // 0x01 = AT命令协议
  pInterface->EPs           = (pInitData->EPInt != 0u) ? (1uL << pInitData->EPInt) : 0uL;
  pInterface->iName         = USB__AllocStringDesc("CDC");
  pInterface->pfOnClassRequest    = _OnClassRequest; // CDC类请求处理回调


  // ========== 第二个接口：CDC DATA 数据接口 ==========
  InterfaceNum              = USB__AllocIF();
  pInterface                = &USB_Global.aIF[InterfaceNum];
  pInterface->pfAddFuncDesc = NULL;
  pInterface->IFClass       = 0x0A;          // 0x0A = CDC Data
  pInterface->IFSubClass    = 0;
  pInterface->IFProtocol    = 0;
  pInterface->EPs           = (1uL << pInitData->EPIn) | (1uL << pInitData->EPOut);
  pInterface->iName         = USB__AllocStringDesc("CDC DATA interface");


// IAD 接口关联描述符（复合设备必备）
  if (USB_Global.pIadAPI != NULL) 
    USB_Global.pIadAPI->pfAdd((U8)FirstInterfaceNo, 2, 2, 2, 0);
  else  
    USB_Global.Class    = 2;
    USB_Global.SubClass = 2;
    USB_Global.Protocol = 0;
   
  // 填充CDC实例上下文
  pInst = &_aInst[_NumInstances];
  pInst->EPIn  = pInitData->EPIn;
  pInst->EPOut = pInitData->EPOut;
  pInst->EPInt = pInitData->EPInt;

  // OUT端点：设置成连续读取模式（随时准备好接收电脑的数据）
  USBD_SetReadMode(pInst->EPOut, USB_READ_MODE_CONTINUOUS);  

  // IN端点：设置成扩展写入模式（支持异步发送大数据）
  USBD_SetWriteMode(pInst->EPIn, USB_WRITE_MODE_EXTEND);     
 
  //初始化一下默认串口参数（波特率等）
  pInst->LineCoding = _DefaultLineCoding; // 默认波特率9600、8N1
  pInst->ControlInterfaceNum = FirstInterfaceNo;
  pInst->DataInterfaceNum    = InterfaceNum;

  USB_LOG((USB_MTYPE_CDC, "CDC: Adding CDC interface with ControlInterface = %d, DataInterface = %d", FirstInterfaceNo, InterfaceNum));
  USB_MEMCPY(pInst->aStatus, aDefaultStatus, sizeof(aDefaultStatus));

  hInst = _NumInstances;
  _NumInstances++;
  return hInst;

_AddEP( pInfo,   pBuffer,  BufferSize)  

  const USB_HW_DRIVER * pDriver;

  //本工厂不提供定时运输服务，会给出拒载提示
  if (pInfo->TransferType == USB_TRANSFER_TYPE_ISO)         
    USB_PANIC("ISO EP added but USB_SUPPORT_TRANSFER_ISO is 0");
   
  pDriver = USB_Global.pDriver;   //USB_Driver_Nordic_nRF52xxx

  if (USB_Global.NumEPs < USB_NUM_EPS)    //1<<8, yes

    pEPStat  = &USB_aEPStat[1];   

    //根据当前USB的传输速度（全速/高速）、通道类型，以及你申请的包大小，重新计算并“阉割/修正”出一个硬件实际支持的最大包长  (待理解，忽略跳过）
    //MaxPacketSize = USB__CalcMaxPacketSize(pInfo->MaxPacketSize, pInfo->TransferType, USB__GetSpeedCapability());
      MaxPacketSize = USB__CalcMaxPacketSize(3072, USB_TRANSFER_TYPE_BULK, USB__GetSpeedCapability());     //nnn

    
    //取得空闲端口1 （从1找到8）
    if (pDriver->pfAllocEPEx != NULL)   //no
       EPDevice = pDriver->pfAllocEPEx(pInfo->InDir, pInfo->TransferType, MaxPacketSize);

    else   
       if ( (pInfo->TransferType == USB_TRANSFER_TYPE_INT) && (MaxPacketSize > 64) )
           USB_PANIC("This driver does not support INT EPs > 64 bytes");        
       EPDevice = pDriver->pfAllocEP(pInfo->InDir, pInfo->TransferType);  //1
     

    if (EPDevice != 0  && EPDevice != 0xFF )  

      USB_MEMSET(pEPStat, 0, sizeof(EP_STAT));
      pEPStat->EPAddr           = (EPDevice) | (pInfo->InDir << 7);    //1 | (0<<7)
      pEPStat->EPType           = pInfo->TransferType;                 //USB_TRANSFER_TYPE_BULK
  
      if ( pInfo->InDir == 0 )         
          
          // 用于OUT的缓冲区大小至少应为 MaxPacketSize，或整数倍          
          if (BufferSize < MaxPacketSize) 
             USB_PANIC("Invalid endpoint buffer size");          
        
        pEPStat->Dir.RxInfo.Buffer.pData = pBuffer;     //usb_cdc_raw_buffer 
        pEPStat->Dir.RxInfo.Buffer.Size  = BufferSize;  //3072

      
      pEPStat->MaxPacketSize      = MaxPacketSize;     //nnn
      pEPStat->MaxPacketSizeUser  = MaxPacketSize;     //nnn
      pEPStat->Interval           = pInfo->Interval;   //0

      r = USB_Global.NumEPs++;

      //把端口编号存到端口地址  （待进一步理解）
      if ( EPDevice < (USB_MAX_ALLOWED_EPADDR + 1) / 2 )   //总共端点数为8，输出输入，除以2后端点数为4
         USB_Global.aPhyAddr2EPIndex[(EPDevice << 1) + pInfo->InDir] = r;  //[2] = 2
 
  return r;

USBD_RegisterSCHook(  * pHook,   *pfStateCb,  * pContext) 
 
  USB_HOOK *  p;

  pHook->cb            = pfStateCb;   //static_usb_on_state_change
  pHook->pContext      = pContext;    //this
   

  //判断是否已重复注册，如重复注册则报错，如果没有则插入到链表里面 (从第一个位置开始，看里面是否已有hook)  //因确定是第一次注册，忽略下面5句
  p = USB_Global.pFirstSCHook;  
  while (p != NULL)  
    if (p == pHook)  
      return 1;     // 重复注册则报错     
    p = p->pNext;    
  
  
  pHook->pNext = USB_Global.pFirstSCHook;
  USB_Global.pFirstSCHook = pHook;            //把hook插入到首位
