
作用：这个 100ms 定时器是 1-Wire 通信的“软件看门狗”，用来检测通信是否卡死，并在卡死时自动恢复


rtc1和rtc2的区别：

   主机还在发数据，但从机软件卡死了	RTC1 CC0（检测到计数器不变）     //100ms
   主机完全不发数据了（断线、断电）	RTC2 CC0（500ms 没收到同步包）   //500ms

步骤	函数	作用
初始化	rtc1_init()	设置分频，配置中断优先级
启动	rtc1_start()	启用 COMPARE0 中断，触发 TASKS_START
软件定时器管理	timer_list_handler()（SWI）	插入/移除定时器节点，更新 CC[0]
超时检测	timer_timeouts_check()	遍历链表，执行到期回调
更新比较值	compare_reg_update()	设置 RTC1->CC[0] 为下一个超时时间
停止	rtc1_stop()	停 RTC1，清空计数器

void comms_uarte_siu_1wire::timed_loop_100ms()
{
    if (state == STATE_UNINITIALIZED)
        return;

    if (last_in_synch_count == in_synch_count &&
        last_lost_synch_count == lost_synch_count &&
        last_acquiring_synch_count == acquiring_synch_count)
    {
 
        state = STATE_ERROR;

        sw_watchdog_count++;

    }
    else
    {
        last_in_synch_count = in_synch_count;
        last_lost_synch_count = lost_synch_count;
        last_acquiring_synch_count = acquiring_synch_count;
    }

    if (state == STATE_ERROR)
    {
        nrfx_timer_compare_int_disable(timer, TIMER_CHANNEL_RX_TIMEOUT);
        if (uarte_resource->rx_in_progress())
        {
 
            uarte_resource->rx_abort();
        }
        else if (uarte_resource->tx_in_progress())
        {
  
            uarte_resource->tx_abort();
        }
        else
        {
            // Synchronize state update with UARTE events
            NRFX_CRITICAL_SECTION_ENTER();
             
            //设置 state 为 STATE_RX_HANDSHAKE
            //停止timer0，关闭ppi_channel_schedule_rx_tx，写入timer0 CC2的阈值64640(4040us)，使能timer0 CC2中断，开启timer0
            //设置rx的接收字符串为 rx_timestamp，长度为5，开启接收
            initiate_handshake_now(TRACE_1WIRE_DISCONNECTED_ERROR_RECOVERY);

            NRFX_CRITICAL_SECTION_EXIT();
        }
    }
}
