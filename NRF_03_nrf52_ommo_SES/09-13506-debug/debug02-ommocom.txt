//设置rx的接收字符串为 rx_timestamp，长度为5，开启接收 （或不等ppi触发）

参数： uarte， &rx_timestamp, 5, false

nrfx_uarte_rx_with_hold(  *p_instance,   p_data,  length,  hold_rx)

    bool second_buffer = false;
 
    p_cb = &m_cb[0];

    // 检查长度是否在范围内，忽略跳过
    NRFX_ASSERT(UARTE_LENGTH_VALIDATE(p_instance->drv_inst_idx, length));  

    // 检查 p_data 指针指向的地址是否在 RAM 范围内  //肯定是，忽略跳过
    if (!nrfx_is_in_ram(p_data))
        return err_code;
    

    //---------------------------------------  buf1和buf2 轮流接收  ------------------------------------------------

    //关闭中断叫快递员先别按门铃（正在洗漱） （有回调函数，说明每次中断你要处理事情，中断是开的。如果没有回调函数，中断没有开，下面就免了）
      不关uart，是在换收货地址为仓库二时，不能中止往仓库一的运货和卸货
      会不会丢失一次ENDRX中断，不会，因为ENDRX事件标志一直都在，一开中断马上就响应了
    if (p_cb->handler)  //comms_uarte_handler    
        nrf_uarte_int_disable(p_instance->p_reg, NRF_UARTE_INT_ERROR_MASK | NRF_UARTE_INT_ENDRX_MASK);  //关闭接收完成中断和err中断
    
    if (p_cb->rx_buffer_length != 0)  //buf1正在使用吗
    
        if (p_cb->rx_secondary_buffer_length != 0) //buf2正在使用吗
        
            if (p_cb->handler)            
                nrf_uarte_int_enable(p_instance->p_reg, NRF_UARTE_INT_ERROR_MASK |   NRF_UARTE_INT_ENDRX_MASK);   //打开接收完成中断和err中断
            
            err_code = NRFX_ERROR_BUSY;        
        second_buffer = true;  // buf2没有使用，请换buf2

    //轮换接收仓库
    if (!second_buffer)  //用buf1
    
        p_cb->rx_buffer_length = length;
        p_cb->p_rx_buffer      = p_data;
        p_cb->rx_secondary_buffer_length = 0;          // 这里两个buf为什么不对称 （这里应该只是初始化）
    
    else   // 用buf2  （这里不能把 buf1的length置为0，因为里面有数据？）
    
        p_cb->p_rx_secondary_buffer = p_data;
        p_cb->rx_secondary_buffer_length = length;


    err_code = NRFX_SUCCESS;

    nrf_uarte_event_clear(p_instance->p_reg, NRF_UARTE_EVENT_ENDRX);    //清ENDRX结束接收事件标志
    nrf_uarte_event_clear(p_instance->p_reg, NRF_UARTE_EVENT_RXTO);     //清RXTO接收超时事件标志

    nrf_uarte_rx_buffer_set(p_instance->p_reg, p_data, length);

    //---------------------------------------  buf1和buf2 轮流接收  ----end--------------------------------------------
 


    //开启接收
    if (!hold_rx)   // yes
    
        if (!second_buffer)        
            nrf_uarte_task_trigger(p_instance->p_reg, NRF_UARTE_TASK_STARTRX);      // 直接启动接收
        
        else        
            nrf_uarte_shorts_enable(p_instance->p_reg, NRF_UARTE_SHORT_ENDRX_STARTRX);    // 设置硬件“自动接力”：当前接收完成后，自动开始下一个接收
        

    if (m_cb[0].handler == NULL)    //no,  comms_uarte_handler  
    
        //等待，直到发生以下三种情况之一才退出循环
        do {
            endrx  = nrf_uarte_event_check(p_instance->p_reg, NRF_UARTE_EVENT_ENDRX);    //接收完成
            rxto   = nrf_uarte_event_check(p_instance->p_reg, NRF_UARTE_EVENT_RXTO);         //接收超时
            error  = nrf_uarte_event_check(p_instance->p_reg, NRF_UARTE_EVENT_ERROR);      //通讯错误

        } while ((!endrx) && (!rxto) && (!error));

        m_cb[0].rx_buffer_length = 0;   //len_buf1 =0   
    
    else    //yes
        nrf_uarte_int_enable(p_instance->p_reg, NRF_UARTE_INT_ERROR_MASK |   NRF_UARTE_INT_ENDRX_MASK);  //打开 UARTE 的“接收完成”和“错误”中断

//清空rx里面的垃圾数据 easy

uarte_basic::uarte_rx_flush_fifo() const
 
    if (nrf_uarte_event_check(uarte_instance->p_reg, NRF_UARTE_EVENT_RXDRDY))  //如果有数据 （垃圾）
 
        uint8_t empty_buff[8];
        nrf_uarte_rx_buffer_set(uarte_instance->p_reg, empty_buff, sizeof(empty_buff));  //准备一个水桶
 
        nrf_uarte_event_clear(uarte_instance->p_reg, NRF_UARTE_EVENT_RXTO);      //清除 rx超时事件标记
        nrf_uarte_event_clear(uarte_instance->p_reg, NRF_UARTE_EVENT_ENDRX);     //清除 rx结束事件标记

        nrf_uarte_task_trigger(uarte_instance->p_reg, NRF_UARTE_TASK_FLUSHRX);   //把水桶里的水，冲到rx里（替代掉原来的垃圾）


        //等待rx被冲洗干净
        uint32_t iter_count = 0;
        while (!nrf_uarte_event_check(uarte_instance->p_reg, NRF_UARTE_EVENT_RXTO) &&    //“冲完了吗？”（RXTO）
               !nrf_uarte_event_check(uarte_instance->p_reg, NRF_UARTE_EVENT_ENDRX) &&   // “rx满了吗”（ENDRX）
               iter_count < 150)                                                         //等太久了，强制退出（防止卡死）
        {
            iter_count++;
        }
         
        nrf_uarte_event_clear(uarte_instance->p_reg, NRF_UARTE_EVENT_RXDRDY);    // 清除rx收到数据标记
 

    //清除rx的err事件标记
    nrf_uarte_event_clear(uarte_instance->p_reg, NRF_UARTE_EVENT_ERROR);

void comms_uarte_siu_1wire::on_comms_uarte_event(nrfx_uarte_event_t const *p_event)
{
    if (p_event->type == NRFX_UARTE_EVT_ERROR)
        TRACE_1WIRE(TRACE_1WIRE_ERROR);
    else if (p_event->type == NRFX_UARTE_EVT_RX_DONE)
        TRACE_1WIRE(TRACE_1WIRE_RX_DONE);
    else if (p_event->type == NRFX_UARTE_EVT_TX_DONE)
        TRACE_1WIRE(TRACE_1WIRE_TX_DONE);

    if (p_event->type == NRFX_UARTE_EVT_ERROR)
    {
        // After aborting rx, event may be ERROR or RX_DONE.  ERROR has priority, and only one of
        // ERROR or RXEND will be processed by the ISR, i.e. if ERROR occurts, RXEND is left
        // pending and causes a second interrupt.
        nrf_uarte_event_clear(uarte->p_reg, NRF_UARTE_EVENT_ENDRX);
    }

    if (state == STATE_ERROR || state == STATE_UNINITIALIZED)
        return;

    if (p_event->type == NRFX_UARTE_EVT_RX_DONE)
    {
        NRFX_ASSERT(!uarte_resource->tx_in_progress());
        NRFX_ASSERT(state >= STATE_FIRST_RX);

        // Handshake packet reception should be handled the same way during handshake restart
        // or randomly receiving a handshake request while connected
        if (state == STATE_RX_RESTART_HANDSHAKE &&
            (rx_timestamp.flags & COMMS_1WIRE_FLAGS_HANDSHAKE))
        {
            state = STATE_RX_TIMESTAMP;
        }

        // Force handshake if the timestamp doesn't make sense
        if (state == STATE_RX_TIMESTAMP &&
            !(rx_timestamp.flags & COMMS_1WIRE_FLAGS_HANDSHAKE) &&
            disconnected_count == 0)
        {
            uint32_t ts_delta = rx_timestamp.timestamp - last_rx_timestamp;
            //             ticks/sec  * sec/ms * ms/synch + 10% tolerance = ~70k ticks/synch
            if (ts_delta > HFCLK_FREQ / 1000   * 4        * 110/100)
            {
#ifdef DEBUG_NRF
                ts_delta_abnormal_count++;
#endif
                on_disconnection(TRACE_1WIRE_DISCONNECTED_BAD_TIMESTAMP);
                state = STATE_RX_RESTART_HANDSHAKE;
            }
#ifdef DEBUG_NRF
            else
            {
                if (ts_delta < ts_delta_min)
                    ts_delta_min = ts_delta;
                if (ts_delta > ts_delta_min)
                    ts_delta_max = ts_delta;
            }
#endif
        }

        // Expecting a handshake packet, but packet timing is unknown?
        if (state == STATE_RX_HANDSHAKE)
        {
            handle_handshake_packet(TRACE_1WIRE_DISCONNECTED_SLAVE_INITIATING_HANDSHAKE);
        }
        // Expecting a handshake packet, but master is transmitting a series of packets?
        else if (state == STATE_RX_RESTART_HANDSHAKE)
        {
            comms_1wire_state_t new_state;
            if (rx_timestamp.flags & COMMS_1WIRE_FLAGS_MASTER_TX)
            {
                // Remain in this state until master permits slave to transmit
                new_state = STATE_RX_RESTART_HANDSHAKE;
            }
            else
            {
                // Deliberately disobey the master's request for slave to transmit to force
                // master to re-initiate the handshake
                new_state = STATE_RX_HANDSHAKE;
            }
            acquiring_synch_count++;
            schedule_rx_timestamp(COMMS_1WIRE_DATA_PACKET_DELAY_TICKS + COMMS_1WIRE_DATA_PACKET_TX_TICKS,
                                  new_state);
        }
        // Expecting a timestamp packet?
        else if (state == STATE_RX_TIMESTAMP)
        {
            // Expected a timestamp packet, but actually got a handshake packet.  Master may have
            // reset or didn't receive last slave transmission.
            if (rx_timestamp.flags & COMMS_1WIRE_FLAGS_HANDSHAKE)
            {
                handle_handshake_packet(TRACE_1WIRE_DISCONNECTED_MASTER_INITIATED_HANDSHAKE);
            }
            else
            {
                in_synch_count++;
                last_rx_timestamp = rx_timestamp.timestamp;
                if (disconnected_count != 0)
                    on_connection();

                uint32_t rxend_timestamp = timestamp_read_capture(timestamp_capture_channel);

#ifdef DEBUG_NRF
                connection_duration++;
                if (connection_duration > connection_max_duration)
                    connection_max_duration = connection_duration;
#endif

                //Process timestamp
                if (rx_timestamp.flags & COMMS_1WIRE_FLAGS_EXT_SYNCH)
                    timestamp_synch_recieved(rx_timestamp.timestamp, rxend_timestamp, TIMESTAMP_SYNCH_SOURCE_1WIRE);

                //RX/TX under the direction of the master
                if (rx_timestamp.flags & COMMS_1WIRE_FLAGS_MASTER_TX)
                {
                    //If the application isn't expecting any data, simply ignore the incoming data,
                    //and schedule receipt of the next timestamp packet
                    if (rx_buffer == nullptr)
                    {
#ifdef DEBUG_NRF
                        lost_packets++;
#endif
                        schedule_rx_timestamp(COMMS_1WIRE_DATA_PACKET_DELAY_TICKS + COMMS_1WIRE_DATA_PACKET_TX_TICKS,
                                              STATE_RX_TIMESTAMP);
                    }
                    else
                    {
                        rx_data_now();
                    }
                }
                else
                {
                    //Master has directed the slave to transmit.  The slave MUST transmit to
                    //maintain a connection.  If the master requested a ping (ex. TEST_PATTERN_MODE
                    //enabled), or if the application has nothing to transmit, then send a ping
                    //packet.
                    if (tx_buffer != nullptr && !(rx_timestamp.flags & COMMS_1WIRE_FLAGS_PING))
                        schedule_tx_data(tx_buffer, STATE_TX_DATA);
                    else
                        schedule_tx_data(&tx_no_data, STATE_TX_PING_ACK);
                }
            }
        }
        // Expecting master to transmit a data packet?
        else if (state == STATE_RX_DATA)
        {
            if (on_rx_complete_in_isr())
                rx_timestamp_now();
            else
                initiate_handshake_now(TRACE_1WIRE_DISCONNECTED_RX_CALLBACK_ERROR);
        }
        else /*if (state == COMMS_1WIRE_STATE_RX_TIMEOUT)*/
        {
            initiate_handshake_now(TRACE_1WIRE_DISCONNECTED_RX_CANCELLED);
        }
    }
    else if (p_event->type == NRFX_UARTE_EVT_TX_DONE)
    {
        NRFX_ASSERT(!uarte_resource->rx_in_progress());
        NRFX_ASSERT(state < STATE_FIRST_RX);

        if (state == STATE_TX_DATA)
            on_tx_complete_in_isr();

        rx_timestamp_now();
    }
    else /*if (p_event->type == NRFX_UARTE_EVT_ERROR)*/
    {
        // Prevent a timeout interrupt while processing a framing error
        nrf_timer_event_clear(timer->p_reg, nrf_timer_compare_event_get(TIMER_CHANNEL_RX_TIMEOUT));

        if (state == STATE_RX_TIMEOUT)
        {
            initiate_handshake_now(TRACE_1WIRE_DISCONNECTED_RX_TIMEOUT);
        }
        else
        {
            if (state >= STATE_FIRST_RX)
            {
                on_disconnection(TRACE_1WIRE_DISCONNECTED_RX_ERROR);

#ifdef DEBUG_NRF
                if (state == STATE_RX_HANDSHAKE)
                    rx_handshake_error_count++;
                else
                    rx_other_error_count++;
#endif

                uarte_resource->uarte_rx_flush_fifo();

                // Framing errors indicate reception was initiated during a timestamp or data
                // packet.  Schedule the next read where it is likely to land in a window where
                // the master is listening for a response from the slave.  Note that the callee
                // adds a bit of time, so that this is not an even multiple of the period.  In
                // other words, future reads will not occur at the exact same time relative to
                // the timestamp packet.
                schedule_rx_timestamp(COMMS_1WIRE_MAX_PERIOD_TICKS / 2, STATE_RX_HANDSHAKE);
            }
            else
            {
#ifdef DEBUG_NRF
                // A framing error is inexplicably reported after the first successful reception
                // following a cold boot
                tx_error_count++;
#endif
            }
        }
    }
}

static void msiu_1wire_packet_received(    *buffer,     length,      *context)
{
    // The 1-wire interface is intended to emulate a USB connection.  The device is connected to an
    // adapter with a USB CDC instance that simply relays USB packets over 1-wire to the device.

    //取得返回长度
    uint32_t response_length = process_usb_serial_packet_received(buffer, length, tx_buffer, PACKET_RESPONSE_BUFFER_SIZE);

    if(response_length == 0) //No response, must be an unknown command
        response_length = fill_in_ack_packet(tx_buffer, OMMO_ACK_UNKNOWN_COMMAND);

    //根据返回长度，发送返回字符串
    comms_1wire.send_command_packet_async(tx_buffer, response_length);

}

uint16_t process_usb_serial_packet_received(uint8_t data[], uint16_t length, uint8_t response_buffer[], uint16_t response_buffer_size)
{
uint16_t response_length = 0;

    //Check for invalid packets
    if(length == 0x00)
        return fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_DATA);

    switch(data[0])
    {
        case OMMO_COMMAND_USB_DATA_ENABLE:
            if(data_mode != DATA_MODE_DISABLED || !sensors_is_state_machine_ready_to_run())
            {
                response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_MODE);
            }
            else if(length != 2)
            {
                response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_DATA);
            }
            else
            {
                //Try to start state machine
                uint32_t rvalue = sensors_start_state_machine();

                if(rvalue == NRF_SUCCESS)
                {
                    packet_id = data[1];
                    data_mode = DATA_MODE_USB;
                }

                //Return success/failure
                set_leds();
                response_length = fill_in_ack_packet(response_buffer, convert_nrf_error_code_to_ommo_ack_code(rvalue));
            }
            break;

        case OMMO_COMMAND_USB_SET_SYNCH_CH_ID:
            if(length != 2)
            {
                response_length = fill_in_ack_packet(response_buffer, OMMO_ACK_INVALID_DATA);
            }
            else
            {
                response_length = fill_in_ack_packet(response_buffer, comms_esb_set_synch_rx_channel(data[1]));
            }
            break;
    }

    if(response_length == 0) //No response, try protocol agnostic commands
        response_length = process_packet_received(data, length, response_buffer, response_buffer_size);

    return response_length;
}

comms_packetizer::send_command_packet_blocking(  * buffer,   length,   block)

    uint16_t tx_write_location;
    ret_code_t err_code = NRF_SUCCESS;
 
    if (transport_device->is_connected())
    {
        // Reserve space in buffer if USB packet fits in USB buffer
        const size_t max_possible_output_length = COBS_CMD_CRC_POST0_MAX_LEN(length);

        CRITICAL_REGION_ENTER();
        if (tx_buffer_active_top + max_possible_output_length < USB_SERIAL_TX_BUFFER_SIZE)
        {
            tx_write_location = tx_buffer_active_top;
            tx_buffer_active_top += max_possible_output_length;
        }
        else
        {
            // Buffer full
            tx_lost_packets++;
            err_code = NRF_ERROR_BUSY;
        }
        CRITICAL_REGION_EXIT();

        // If data fit into the buffer, copy into reserved buffer location
        // Note: usb event queue processing is done at the lowest priority level so this process
        // cannot be interrupted
        if (err_code == NRF_SUCCESS)
        {
            cobs_encode_usb_command_crc_post0_zero_pad(buffer, length, tx_buffer_active + tx_write_location, max_possible_output_length);

            err_code = flush_tx_buffers(block);
        }
    }
    else    
        tx_lost_packets++;
