    // 检查是不是有效的握手包  timestamp = 0xfffefdfc 和 flags.7 =1
    if (rx_timestamp.timestamp == COMMS_1WIRE_TIMESTAMP_DURING_HANDSHAKE &&  (rx_timestamp.flags & COMMS_1WIRE_FLAGS_HANDSHAKE))  //是有效握手包，如下处理
    
        //在log中标记当前已断开，并记录断开的次数
        on_disconnection(reason);

        // flags.0,1 = 1,1  满足完整的握手
        if (!(rx_timestamp.flags & COMMS_1WIRE_FLAGS_MASTER_TX) &&  (rx_timestamp.flags & COMMS_1WIRE_FLAGS_PING))

        initiate_handshake_now(reason);
