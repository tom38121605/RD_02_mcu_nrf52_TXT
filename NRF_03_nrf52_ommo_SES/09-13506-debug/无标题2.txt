
通讯过程：
连接建立	握手包，	连续发送，直到收到从机响应	从机回空包后进入正常通信
正常通信（有数据），	时间戳包 + 数据包	每 4ms 一个周期	先发 timestamp，300us 后发数据包
正常通信（无数据），	时间戳包	每 4ms 一个周期	只发 timestamp，不发数据包



//在STATE_RX_RESTART_HANDSHAKE步骤，如果flags.7=1是握手包，则state向前进一步到  STATE_RX_TIMESTAMP   
//在STATE_RX_TIMESTAMP步骤，如果flags.7=0不是握手包。判断跟上一个包的时间相隔太久，如果太久则需要重新握手 （中止解释时间戳的包）


问1：第一次握手的时候，rx_buffer = nullptr。请问 之后在哪里给rx_buffer赋值的

答： 第一次运行时 rx_buffer == nullptr 是正常的，因为 packetizer.enable_packet_reading() 还没有被调用。 
     当你收到 timestamp 包时，on_connection() 触发了 packetizer_enabled_target = true，但主循环还没来得及执行 enable_packet_reading()，所以 rx_buffer 还是 nullptr。
     之后，rx_buffer 就在 comms_transport_rx() 中被赋值了。完整流程如下：

      [收到 timestamp 包]
          ↓
      on_connection()
          ↓
      packetizer_enabled_target = true
          ↓
      schedule_rx_timestamp(..., STATE_RX_TIMESTAMP)
          ↓
      [等待 3.53ms 后收下一个 timestamp 包]
          ↓
      （在这 3.53ms 期间，主循环执行）
          ↓
      event_queue_packetizer_enabled_change_process()
          ↓
      packetizer.enable_packet_reading()
          ↓
      comms_transport_rx(m_rx_buffer, ...)
          ↓
      rx_buffer = m_rx_buffer   // ★ 赋值 ★
          ↓
      [3.53ms 后，收到下一个 timestamp 包]
          ↓
      MASTER_TX = 1
          ↓
      检查 rx_buffer != nullptr → 可以收数据包了
          ↓
      rx_data_now()

comms_uarte_siu_1wire::on_rx_complete_in_isr()

    //在编译时检查 data 成员是否位于结构体的起始位置（偏移量为 0），在结构体开头可以简化地址计算 (防止其他人修改成员位置)
    STATIC_ASSERT(offsetof(comms_1wire_master_tx_data_packet_t, data) == 0);

    //取得接收缓冲区的结构体指针和数据长度
    packet = (comms_1wire_master_tx_data_packet_t*)rx_buffer;    //结构体指针
    actual_length = packet->length;                              //接收数据部分的长度


    //下面两句仅用作调试，忽略跳过
    if (actual_length == 0)
        verify_test_pattern(packet);

    rx_buffer = nullptr;    //清空指针rx_buffer


    //如果长度超限，返回false
    if (actual_length > COMMS_1WIRE_DATA_PACKET_MAX_SIZE)   //320    
        //rx_complete_callback(COMMS_1WIRE_DATA_PACKET_MAX_SIZE, true, p_context); 
           rx_failed = true;
        return false;

    //把推车rx_buffer里的数据，搬到仓库rx_cobs_buffer里
    //如果没有断连，就继续接收数据包
    //继续设置rx_buffer为：数据包的变量rx_raw_buffer
    rx_complete_callback(actual_length, false, p_context);  //参考： debug15_packetizer.txt

    return true;

timestamp_synch_recieved(   basestation_ts,  local_event_ts,   source)

    tx_delay = 0;

    timestamp_synch_received_msk = 0xFF;

    switch(source)

        case TIMESTAMP_SYNCH_SOURCE_1WIRE:

            tx_delay = TIMESTAMP_SYNCH_1WIRE_TX_DELAY;   //813

            
                //写入rtc2 CC0的阈值 “RTC2 当前计数值 + 上次设置的延时时间500ms”，使能rtc2 CC0中断
                //timestamp_synch_lost = false如果有超时标记，则去掉超时标记。设置小灯状态为未超时正常状态
                timestamp_reset_synch_lost_timer();
  

                //-----------------关于无线同步和有线同步的切换-----忽略跳过------------------

                //was_wireless = !timestamp_is_synch_wired();                   //没切换前（初始）是1， 是无线sync
                     was_wireless = !(timestamp_synch_esb_ignore_count > 0);    //timestamp_synch_esb_ignore_count大于0，表示是有线

                timestamp_synch_esb_ignore_count = 16    //设置为有线

                if(timestamp_synch_lost_found_callback != NULL && was_wireless)         //如果是无线  (上层关心同步状态变化，且之前是无线模式)
  
                    //timestamp_synch_lost_found_callback(!timestamp_synch_lost, true);  //通知上层：“同步恢复了，现在用的是有线同步”
                        set_leds();


                //-----------------关于无线同步和有线同步的切换--------end---------------
            
            break;


    //-1 时间补偿
    basestation_ts -= 1;


    //保存上一次的同步误差 
    last_timestamp_synch_offset = timestamp_synch_offset;

    //计算本次新的同步误差 
    timestamp_synch_offset = (basestation_ts + tx_delay) - local_event_ts;   //easy，16000-3000=13000


    //如果这次的偏移突然比上次小了超过 1000 ticks 
    if(last_timestamp_synch_offset - timestamp_synch_offset > 1000)    
        __NOP();                                                                                           //（或为备用调试）
    

    //计算timer3 cc0下一次的门槛值           //应可优化成： local_event_ts + (offsetold - offsetnew) + 1ms
    sample_event_trigger_value = basestation_ts + OMMO_TIMESTAMP_SAMPLE_PERIOD - timestamp_synch_offset;   //15984

    //下面判断是否更新timer3 cc1的同步时间门槛
    AAA = basestation_ts - last_synch_event_value_bs_units;
    BBB = OMMO_TIMESTAMP_SAMPLE_PERIOD*(OMMO_TIMESTAMP_SYNCH_PERIOD_MULT-1);      // 15984 *（4-1）

    if( AAA > BBB ) //如果是第4个1ms的同步信号，则重新设置timer3 cc1的同步时间门槛
    
        last_synch_event_value_bs_units = basestation_ts;  //保存主机时间

        //计算timer3 cc1下一次的门槛值
        synch_event_trigger_value = basestation_ts + OMMO_TIMESTAMP_SYNCH_PERIOD - timestamp_synch_offset;

        //写入timer3 CC1的阈值 当前计数值local_event_ts + 15984*4 (4ms)，使能timer3 CC1中断  //cc1做同步用
        nrfx_timer_compare(&timer_timestamp, timestamp_synch_event_compare_channel, synch_event_trigger_value, false);

//总功能：算好时间 → 到点了硬件自动开始收 → 没收完就超时报警
//取得timer0 cc0的计数值，即rxend的时间点
//写入timer0 CC1的阈值 rxend_time + data_end_offset(3530us) +3760(235us)，不使能timer0 CC1中断
//设置ppi：ppi_channel_schedule_rx_tx(闹铃)  -->  startrx(开灯) ，使能ppi
//写入timer0 CC2的阈值 rxend_time + data_end_offset（3530us) + 8640(540us)，使能timer0 CC2中断  
//设置rx的接收字符串p_rx_buffer为 rx_timestamp，长度为5，暂不接收(等ppi触发)
//设置state为new_state

参数： 56480 ， RESTART_HANDSHAKE / RX_HANDSHAKE

comms_uarte_siu_1wire::schedule_rx_timestamp(  data_end_offset,  new_state )

    state = new_state;

    //取得timer0 cc0的计数值，即rxend的时间点
    rxend_time = nrf_timer_cc_read(timer->p_reg, TIMER_CHANNEL_CAPTURE_RXEND);

    //写入timer0 CC1的阈值 rxend_time + data_end_offset/3530us +3760/235us，不使能timer0 CC1中断 （ 直接触发ppi_channel_schedule_rx_tx，ppi再触发startrxrx ）
    nrfx_timer_compare(timer, TIMER_CHANNEL_SCHEDULE_RX_TX,
                       rxend_time + data_end_offset + COMMS_1WIRE_TIMESTAMP_PACKET_MAX_DELAY_TICKS / 2, false);

    //设置ppi：ppi_channel_schedule_rx_tx(闹铃)  -->  startrx(开灯) 
    nrf_ppi_task_endpoint_setup(ppi_channel_schedule_rx_tx, uarte_resource->get_startrx_task());

    //使能ppi
    nrfx_ppi_channel_enable(ppi_channel_schedule_rx_tx);


    //在定时器（TIMER0）上设置一个“接收超时闹钟”——如果从预期的 timestamp 包开始接收后，超过 540us 还没收到完整数据，就触发超时中断
    //写入timer0 CC2的阈值 rxend_time + data_end_offset/3530us +8640/540us，使能timer0 CC2中断  
    nrfx_timer_compare(timer, TIMER_CHANNEL_RX_TIMEOUT,
                       rxend_time + data_end_offset + COMMS_1WIRE_TIMESTAMP_PACKET_MAX_DELAY_TICKS +
                       COMMS_1WIRE_TIMESTAMP_PACKET_TX_TICKS + COMMS_1WIRE_RX_TIMEOUT_TICKS,
                       true);


    //设置rx的接收字符串p_rx_buffer为 rx_timestamp，长度为5，暂不接收(等ppi触发)
    //triggered_rx(&rx_timestamp, sizeof(rx_timestamp));
       rx_with_optional_hold(buffer, length, true)

//双重检查是不是有效的握手包：  数据timestamp = 0xfffefdfc 和 标记flags.7 =1。否则  -- >  重新开始握手 （一直开startrx等待数据），state保持为STATE_RX_HANDSHAKE
//再检查主机要求从机返回0长度数据包，则如下处理。 否则  -- >  再次握手 （3.5ms+后开startrx等待数据），state保持为STATE_RX_HANDSHAKE
//上面都检查通过，则规划uart的发送空ack数据, 改变state为TATE_TX_PING_ACK

参数： TRACE_1WIRE_DISCONNECTED_SLAVE_INITIATING_HANDSHAKE    /B2

comms_uarte_siu_1wire::handle_handshake_packet( reason)
 
    //双重检查是不是有效的握手包：  timestamp = 0xfffefdfc 和 flags.7 =1
    AAA = rx_timestamp.timestamp == COMMS_1WIRE_TIMESTAMP_DURING_HANDSHAKE;    //数据 timestamp=0xfffefdfc, 是握手包
    BBB = rx_timestamp.flags & COMMS_1WIRE_FLAGS_HANDSHAKE ;                   //标记 flag.7=1,是握手包
    if ( AAA && BBB )  //是有效握手包，如下处理
    
        //在log中标记当前已断开，并记录断开的次数
        on_disconnection(reason);  //原因：正在握手中

        // flags.0,1 = 1,1  满足完整的握手
        CCC = !(rx_timestamp.flags & COMMS_1WIRE_FLAGS_MASTER_TX);    //标记 flag.0=0,不是主机发送数据
        DDD = rx_timestamp.flags & COMMS_1WIRE_FLAGS_PING;            //标记 flag.1=1,从机将返回0长度数据包 

        //如果不是主机发送数据，要求从机返回0长度数据包，则如下处理
        if ( CCC &&  DDD )
         
            //总功能：规划uart的发送空ack数据
            //设置state为TATE_TX_PING_ACK 
            //取得timer0 cc0的计数值，即rxend的时间点
            //写入timer0 CC1的阈值4800(300us)，不使能timer0 CC1中断 （后面会直接触发ppi）
            //建立ppi：ppi_channel_schedule_rx_tx(闹铃)   -->  starttx(开灯) ，使能ppi
            //关闭timer0 cc2的超时中断
            //---把单线引脚从rx变成tx
            //把323字节的发送数据tx_no_data放到uart的tx的缓冲区，暂不发送，或等ppi触发
            schedule_tx_data(&tx_no_data, STATE_TX_PING_ACK);
         

        else  //握完手，但主机没让干活，我就继续等着
         
            acquiring_synch_count++;


            //总功能：算好时间 → 到点了硬件自动开始收 → 没收完就超时报警
            //取得timer0 cc0的计数值，即rxend的时间点
            //写入timer0 CC1的阈值 rxend_time + data_end_offset(3530us) +3760(235us)，不使能timer0 CC1中断
            //设置ppi：ppi_channel_schedule_rx_tx(闹铃)  -->  startrx(开灯) ，使能ppi
            //写入timer0 CC2的阈值 rxend_time + data_end_offset(3530us) + 8640(540us)，使能timer0 CC2中断  
            //设置rx的接收字符串p_rx_buffer为 rx_timestamp，长度为5，暂不接收(等ppi触发)
            //设置state为 STATE_RX_HANDSHAKE
            schedule_rx_timestamp(COMMS_1WIRE_DATA_PACKET_DELAY_TICKS + COMMS_1WIRE_DATA_PACKET_TX_TICKS, STATE_RX_HANDSHAKE);   //56480 (3530us)
         
    

    else   //不是有效握手包，处理 如下

         //设置 state 为 STATE_RX_HANDSHAKE
        //停止timer0，关闭ppi_channel_schedule_rx_tx，写入timer0 CC2的阈值64640(4040us)使能timer0 CC2中断，开启timer0
        //设置rx的接收字符串为 rx_timestamp，长度为5，开启接收 
        initiate_handshake_now(reason);

comms_uarte_siu_1wire::schedule_tx_data(   *buffer,  new_state)
 
    state = new_state;

comms_uarte_siu_1wire::rx_timestamp_now( )

    //停止timer0定时器
    nrfx_timer_pause(timer);

    nrfx_ppi_channel_disable(ppi_channel_schedule_rx_tx);
