//停止timer0，关闭ppi_channel_schedule_rx_tx，写入timer0 CC2的阈值64640(4040us)使能timer0 CC2中断，开启timer0

参数：TRACE_1WIRE_DISCONNECTED_INITIAL_STATE

comms_uarte_siu_1wire::initiate_handshake_now(  reason )

    lost_synch_count++;
    state = STATE_RX_HANDSHAKE;

    //在log中标记当前已断开，并记录断开的次数
    on_disconnection(reason);

    //stop停止timer0计数器
    nrfx_timer_pause(timer);

    //关闭ppi_channel_schedule_rx_tx，不让它触发
    nrfx_ppi_channel_disable(ppi_channel_schedule_rx_tx);

    //写入timer0 CC2的阈值64640(4040us)，使能timer0 CC2中断
    nrfx_timer_compare(timer, TIMER_CHANNEL_RX_TIMEOUT, COMMS_1WIRE_MAX_PERIOD_TICKS + COMMS_1WIRE_RX_TIMEOUT_TICKS, true);   // 64640

    nrfx_timer_clear(timer);    //计数器清0
    nrfx_timer_resume(timer);   //重新计数

    //设置rx的接收字符串为 rx_timestamp，长度为5，开启接收
    //rx_now(&rx_timestamp, sizeof(rx_timestamp));
       rx_with_optional_hold(&rx_timestamp, 5, false)

//取得timer0 cc0的计数值，即rxend的时间点
//写入timer0 CC1的阈值 rxend_time + data_end_offset +7520，不使能timer0 CC1中断
//设置ppi：ppi_channel_schedule_rx_tx(闹铃)  -->  startrx(开灯) ，使能ppi

//设置rx的接收字符串p_rx_buffer为 rx_timestamp，长度为5，暂不接收(等ppi触发)

参数： 56000 ， RESTART_HANDSHAKE / RX_HANDSHAKE

comms_uarte_siu_1wire::schedule_rx_timestamp(  data_end_offset,  new_state )

    state = new_state;

    //取得timer0 cc0的计数值，即rxend的时间点
    rxend_time = nrf_timer_cc_read(timer->p_reg, TIMER_CHANNEL_CAPTURE_RXEND);

    //写入timer0 CC1的阈值 rxend_time + data_end_offset +7520，不使能timer0 CC1中断 （或后面由ppi触发）
    nrfx_timer_compare(timer, TIMER_CHANNEL_SCHEDULE_RX_TX,
                       rxend_time + data_end_offset + COMMS_1WIRE_TIMESTAMP_PACKET_MAX_DELAY_TICKS / 2, false);

    //ppi_channel_schedule_rx_tx(闹铃)  -->  startrx(开灯) 
    nrf_ppi_task_endpoint_setup(ppi_channel_schedule_rx_tx, uarte_resource->get_startrx_task());

    //使能ppi
    nrfx_ppi_channel_enable(ppi_channel_schedule_rx_tx);


    //在定时器（TIMER0）上设置一个“接收超时闹钟”——如果从预期的 timestamp 包开始接收后，超过 540us 还没收到完整数据，就触发超时中断
    //写入timer0 CC2的阈值 rxend_time + data_end_offset +8640(540us)，使能timer0 CC2中断  
    nrfx_timer_compare(timer, TIMER_CHANNEL_RX_TIMEOUT,
                       rxend_time + data_end_offset + COMMS_1WIRE_TIMESTAMP_PACKET_MAX_DELAY_TICKS +
                       COMMS_1WIRE_TIMESTAMP_PACKET_TX_TICKS + COMMS_1WIRE_RX_TIMEOUT_TICKS,
                       true);


    ////设置rx的接收字符串为 rx_timestamp，长度为5，开启接收
    //triggered_rx(&rx_timestamp, sizeof(rx_timestamp));
       rx_with_optional_hold(buffer, length, true)

comms_uarte_siu_1wire::schedule_tx_data(   *buffer,  new_state)
 
    state = new_state;

    //取得timer0 cc0的计数值，即rxend的时间点
    //rxend_time = nrf_timer_cc_read(timer->p_reg, TIMER_CHANNEL_CAPTURE_RXEND);   //TIMER0 CC0
       rxend_time = NRF_TIMER0->CC0;

    //写入timer0 CC1的阈值4800(300us)，不使能timer0 CC1中断 （后面会直接触发ppi）
    nrfx_timer_compare( timer, TIMER_CHANNEL_SCHEDULE_RX_TX,  rxend_time + COMMS_1WIRE_DATA_PACKET_DELAY_TICKS,  false );


    //建立ppi：ppi_channel_schedule_rx_tx(闹铃)   -->  starttx(开灯) 
    nrf_ppi_task_endpoint_setup(ppi_channel_schedule_rx_tx, uarte_resource->get_starttx_task());

    //使能ppi_channel_schedule_rx_tx
    nrfx_ppi_channel_enable(ppi_channel_schedule_rx_tx);


    //禁用timer0 cc2的超时中断（发送期间不需要超时检测）
    nrfx_timer_compare_int_disable(timer, TIMER_CHANNEL_RX_TIMEOUT);


    //把单线引脚从rx变成tx
    //把发送数据放到uart的tx的缓冲区，暂不发送，或等ppi触发
    tx_with_hold(buffer, sizeof(comms_1wire_slave_tx_data_packet_t));

void timestamp_add_capture_event_with_disable(uint32_t trigger_event_addr, nrf_timer_cc_channel_t capture_channel)
{
nrf_ppi_channel_t ppi_channel_timestamp_capture;

    //Setup PPI for capturing timestamp at encoder edge event
    APP_ERROR_CHECK(nrfx_ppi_channel_alloc(&ppi_channel_timestamp_capture));
    APP_ERROR_CHECK(nrfx_ppi_channel_assign(ppi_channel_timestamp_capture, trigger_event_addr, nrfx_timer_task_address_get(&timer_timestamp, capture_tasks[capture_channel])));
    APP_ERROR_CHECK(nrfx_ppi_channel_enable(ppi_channel_timestamp_capture));

    //Include in disable group
    APP_ERROR_CHECK(nrfx_ppi_channel_include_in_group(ppi_channel_timestamp_capture, ppi_group_disable_timestamp_capture));
}
