comms_packetizer::transport_read_complete(  actual_rx_length,  failed)

    uint16_t i_raw;
    uint16_t i_cobs;


    //如果是失败标记，忽略跳过
    if (failed)
        rx_failed = true;   return;
 
    //嵌套调用出错处理，忽略跳过
    //OMMO_APP_ERROR_CHECK(in_comms_transport_rx ? NRF_ERROR_INTERNAL : NRF_SUCCESS, STRING("unexpected synchronous call of transport_read_complete"), 0);
        if(in_comms_transport_rx)  
            app_error_save_and_stop()   //系统停止/复位 


    //判断仓库rx_cobs_buffer里面的空间是否足够。把推车rx_buffer里的数据，搬到仓库rx_cobs_buffer里
    if (rx_cobs_buffer_top + actual_rx_length > sizeof(rx_cobs_buffer))  //空间不够，则不搬
    
        rx_lost_packets++;
        rx_cobs_buffer_top = rx_cobs_buffer_in_use;
    
    else  //空间足够
    
        memcpy(&rx_cobs_buffer[rx_cobs_buffer_top], rx_raw_buffer, actual_rx_length);  //把卡车rx_buffer里的数据，搬到仓库rx_cobs_buffer里
        rx_cobs_buffer_top += actual_rx_length;

        rx_data_ready = true;
    

    //判断是否断连了
    //AAA = transport_device->is_connected();
    //  AAA = comms_uarte_siu_1wire::is_connected();

    //如果没有断连，就继续接收数据包
    //继续设置rx_buffer为：数据包的变量rx_raw_buffer
    if (AAA) 
        //setup_next_rx();
            in_comms_transport_rx = true;

#define L(x) TYPEID(x), x

全局参数：   rx_cobs_buffer

comms_packetizer::event_queue_start_rx_process()


    size_t  cobs_size, , packet_size, not_used;

    AAA = rx_cobs_buffer_in_use ? NRF_ERROR_INTERNAL : NRF_SUCCESS;
    BBB =  L(rx_cobs_buffer_in_use), L(rx_cobs_buffer_bottom), L(rx_cobs_buffer_top), L(rx_lost_packets), L(tx_lost_packets);

    //检查 rx_cobs_buffer_in_use 是否为非零，如果是，触发错误处理并记录相关变量值
     if (AAA)  //NRF_ERROR_INTERNAL
       
        //OMMO_APP_ERROR_CHECK(AAA,BBB, 0);
            L(rx_cobs_buffer_in_use),       // 记录变量,      L(x) 把变量名和变量值打包成一对 (类型, 值)
            L(rx_cobs_buffer_bottom),   // 记录变量
            L(rx_cobs_buffer_top),          // 记录变量
            L(rx_lost_packets),               // 记录变量
            L(tx_lost_packets),                // 记录变量
   
    current_top                        =   rx_cobs_buffer_top;
    rx_cobs_buffer_in_use              =   current_top;


    // 在rx_cobs_buffer 中查找 \0 字符，用来确定一个 COBS 编码包的结束位置
    for (i = rx_cobs_buffer_bottom; i < current_top; i++)
    
        if (rx_cobs_buffer[i] == '\0')
            break;

    cobs_size=i;

    // Bail if this is a partial packet  如果是不完整数据包，则直接退出处理。
    if (cobs_size >= current_top)
    
        rx_cobs_buffer_bottom = current_top;
        rx_cobs_buffer_in_use = 0;
        return;
    

  
    // Decode our rx cobs buffer, in-place
    if (!cobs_decode_crc(
          rx_cobs_buffer, cobs_size,
          rx_cobs_buffer, cobs_size, &crc, &not_used,
          &packet_size))
    {
        packet_size = 0;
    }

    // \0 is excluded for decoding the COBS data, but included in subsequent pointer math
    cobs_size++;

    // Prevent ISR from updating the decoded packet.  If it needs to overwrite data, anything
    // after this point is disposable.
    rx_cobs_buffer_in_use = cobs_size;

    // Is this a real packet?
    if (packet_size > 0x00)
    {
        if (crc == 0x00)
        {
            if (packet_handler)
                (*packet_handler)(rx_cobs_buffer, packet_size, packet_handler_context);
        }
        else
        {
            //Report a 0x00 length packet, indicating an invalid packet was received
            if (packet_handler)
                (*packet_handler)(nullptr, 0x00, packet_handler_context);
        }
    }

    // Shift the processed packet to the start of the cobs buffer, and release use of the buffer
    // to transport_read_complete/ISR.  Note that there may be another packet between cobs_size
    // and current_top.



    // disable_packet_reading() may have been called by the handler
    if (rx_cobs_buffer_in_use != 0)
    {
        NRFX_ASSERT(rx_cobs_buffer_top >= cobs_size);
        memmove(rx_cobs_buffer, &rx_cobs_buffer[cobs_size], rx_cobs_buffer_top - cobs_size);
        rx_cobs_buffer_top -= cobs_size;
        rx_cobs_buffer_in_use = 0;
        rx_cobs_buffer_bottom = 0;
        if (rx_cobs_buffer_top == 0)
            rx_data_ready = false;
    }

static void msiu_1wire_packet_received(    *buffer,     length,      *context)
{
    // The 1-wire interface is intended to emulate a USB connection.  The device is connected to an
    // adapter with a USB CDC instance that simply relays USB packets over 1-wire to the device.

    //取得返回长度
    uint32_t response_length = process_usb_serial_packet_received(buffer, length, tx_buffer, PACKET_RESPONSE_BUFFER_SIZE);

    if(response_length == 0) //No response, must be an unknown command
        response_length = fill_in_ack_packet(tx_buffer, OMMO_ACK_UNKNOWN_COMMAND);

    //根据返回长度，发送返回字符串
    comms_1wire.send_command_packet_async(tx_buffer, response_length);

}
